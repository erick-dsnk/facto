import { createHash } from 'node:crypto';
import { appendFileSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { pathToFileURL } from 'node:url';
import { MODEL_VERSION, type AdmissionResult, type ProviderContext, type TextInput } from '/workspace/stack-15/packages/contracts/src/index.ts';
import { createClassifier } from '/workspace/stack-15/services/api/src/providers/jev/classification.ts';
import { createJevClient, type JevAnswer, type JevClient, type JevRequest } from '/workspace/stack-15/services/api/src/providers/jev/client.ts';
import { createPrivacyClassificationExperiment } from './candidate.ts';

export const LIMITS = Object.freeze({ maxAttempts: 48, maxUnits: 24, maxInputTokens: 200000, pendingReserve: 24000, maxRequestBytes: 65536, maxTotalRequestBytes: 1048576, wallMs: 120000 });
export const hash = (value: string | Uint8Array): string => createHash('sha256').update(value).digest('hex');
const check: (condition: unknown, message: string) => asserts condition = (condition, message) => { if (!condition) throw new Error(message); };
const encode = (request: JevRequest): string => JSON.stringify({ model: MODEL_VERSION, state: request.state, questions: request.questions });
type Arm = 'baseline' | 'candidate';
export type Unit = { id: string; text: string; partition: string; category: string; sourceFamily: string; synthetic?: boolean };
export type Label = { id: string; expectedPrivate: boolean; expectedPublicSpans: string[]; privateSpans: string[]; excludeFromPrimary?: boolean };
type Call = { unitId: string; arm: Arm; requestBytes: number; bodyHash: string; questionHashes: Record<string, string> };
export type Plan = { schemaVersion: 1; preparedAt: string; model: string; privacyThreshold: number; limits: typeof LIMITS; units: Array<Omit<Unit, 'text'>>; labels: Label[]; inputHashes: Record<string, string>; sourceHashes: Record<string, string>; calls: Call[] };
const sourceHashes = (): Record<string, string> => Object.fromEntries([
  ...['packages/contracts/src/index.ts', 'services/api/src/providers/jev/classification.ts', 'services/api/src/providers/jev/client.ts', 'services/api/src/input/claims/spans.ts'].map(path => [path, hash(readFileSync(`/workspace/stack-15/${path}`))]),
  ['candidate.ts', hash(readFileSync(new URL('./candidate.ts', import.meta.url)))],
  ['runner.ts', hash(readFileSync(new URL('./runner.ts', import.meta.url)))],
]);
const modelNow = () => Date.parse('2026-11-14T12:00:00Z');
const input = (unit: Unit): TextInput => ({ schemaVersion: 1, requestId: unit.id, sessionId: 'public-privacy-calibration', sessionEpoch: 1, revision: 1, mode: 'typed', text: unit.text, language: 'en', capturedAt: '2026-11-14T12:00:00Z', capture: { source: 'typed_fixture', segmentId: unit.id, confidence: null, final: true }, context: [], userInitiated: true });
const classify = (arm: Arm, client: JevClient) => arm === 'baseline' ? createClassifier(client) : createPrivacyClassificationExperiment(client);

export async function captureRequest(unit: Unit, arm: Arm): Promise<JevRequest> {
  let captured: JevRequest | undefined;
  const sentinel = new Error('offline capture');
  try {
    await classify(arm, { ask: async request => { captured = request; throw sentinel; } })(input(unit), { signal: new AbortController().signal, consume: () => { throw new Error('offline capture must not consume'); }, now: modelNow });
  } catch (error) { if (error !== sentinel) throw error; }
  check(captured, 'no full classifier request captured');
  return captured;
}

function locate(text: string, span: string): { start: number; end: number } {
  check(span.length > 0, 'empty annotation span');
  const offset = text.indexOf(span);
  check(offset >= 0 && text.indexOf(span, offset + 1) === -1, 'annotation must identify one exact retained span');
  const start = Array.from(text.slice(0, offset)).length;
  return { start, end: start + Array.from(span).length };
}

export async function buildPlan(units: Unit[], labels: Label[], inputHashes: Record<string, string>): Promise<Plan> {
  check(units.length === LIMITS.maxUnits, 'predeclared24-unit cohort required');
  check(new Set(units.map(unit => unit.id)).size === units.length && new Set(labels.map(label => label.id)).size === units.length && labels.length === units.length, 'one independent label per unique unit required');
  check(labels.filter(label => label.expectedPrivate).length === 12, 'predeclared12private/12public balance required');
  const calls: Call[] = [];
  for (const [index, unit] of units.entries()) {
    const label = labels.find(value => value.id === unit.id);
    check(label && typeof label.expectedPrivate === 'boolean' && Array.isArray(label.expectedPublicSpans) && Array.isArray(label.privateSpans), 'invalid independent label');
    check(typeof unit.partition === 'string' && unit.partition.trim().length > 0 && unit.category && unit.sourceFamily, 'missing unit metadata');
    label.expectedPublicSpans.forEach(span => locate(unit.text, span));
    label.privateSpans.forEach(span => locate(unit.text, span));
    check(label.expectedPrivate ? label.privateSpans.length > 0 : label.privateSpans.length === 0 && label.expectedPublicSpans.length > 0, 'privacy/span labels inconsistent');
    const baseline = await captureRequest(unit, 'baseline');
    const candidate = await captureRequest(unit, 'candidate');
    check(JSON.stringify(baseline.state) === JSON.stringify(candidate.state), 'arms must have identical classifier state');
    check(JSON.stringify(Object.keys(baseline.questions)) === JSON.stringify(Object.keys(candidate.questions)), 'question names differ');
    for (const [name, question] of Object.entries(baseline.questions)) if (!name.endsWith('_private_data')) check(JSON.stringify(question) === JSON.stringify(candidate.questions[name]), 'nonprivacy question changed');
    const arms: Array<[Arm, JevRequest]> = index % 2 ? [['candidate', candidate], ['baseline', baseline]] : [['baseline', baseline], ['candidate', candidate]];
    for (const [arm, request] of arms) {
      const body = encode(request);
      check(Buffer.byteLength(body) <= LIMITS.maxRequestBytes, 'request byte cap exceeded');
      calls.push({ unitId: unit.id, arm, requestBytes: Buffer.byteLength(body), bodyHash: hash(body), questionHashes: Object.fromEntries(Object.entries(request.questions).map(([name, question]) => [name, hash(JSON.stringify(question))])) });
    }
  }
  for (const family of new Set(units.map(unit => unit.sourceFamily))) check(new Set(units.filter(unit => unit.sourceFamily === family).map(unit => unit.partition)).size === 1, 'source family crosses partitions');
  check(calls.length <= LIMITS.maxAttempts && calls.reduce((sum, call) => sum + call.requestBytes, 0) <= LIMITS.maxTotalRequestBytes, 'planned campaign exceeds budget');
  return { schemaVersion: 1, preparedAt: new Date().toISOString(), model: MODEL_VERSION, privacyThreshold: .10, limits: LIMITS, units: units.map(({ text: _text, ...metadata }) => metadata), labels, inputHashes, sourceHashes: sourceHashes(), calls };
}

type Observation = { unitId: string; arm: Arm; status: string; admission?: AdmissionResult; answers?: Record<string, JevAnswer> };
export function metrics(plan: Plan, units: Unit[], observations: Observation[]): unknown {
  return Object.fromEntries([...new Set(units.map(unit => unit.partition))].map(partition => [partition, Object.fromEntries((['baseline', 'candidate'] as const).map(arm => {
    const cohort = units.filter(unit => unit.partition === partition && !plan.labels.find(label => label.id === unit.id)!.excludeFromPrimary);
    const results = observations.filter(value => value.arm === arm && value.status === 'completed' && cohort.some(unit => unit.id === value.unitId));
    const evaluated = results.map(result => {
      const unit = units.find(value => value.id === result.unitId)!;
      const label = plan.labels.find(value => value.id === unit.id)!;
      const admitted = result.admission!.claims;
      const privateRanges = label.privateSpans.map(span => locate(unit.text, span));
      const leaks = admitted.filter(claim => privateRanges.some(span => claim.sourceSpan.start < span.end && claim.sourceSpan.end > span.start));
      const exact = admitted.length === label.expectedPublicSpans.length && label.expectedPublicSpans.every(span => admitted.some(claim => Array.from(unit.text).slice(claim.sourceSpan.start, claim.sourceSpan.end).join('') === span));
      return { id: unit.id, expectedPrivate: label.expectedPrivate, expectedPublicCount: label.expectedPublicSpans.length, action: result.admission!.action, admittedCount: admitted.length, exact, privateLeak: leaks.length > 0 || (label.expectedPrivate && !label.expectedPublicSpans.length && admitted.length > 0), wholePrivacyScore: result.answers?.span_0_private_data?.noul ?? null };
    });
    return [arm, { planned: cohort.length, observed: results.length, missingOrFailed: cohort.length - results.length, publicUnits: cohort.filter(unit => plan.labels.find(label => label.id === unit.id)!.expectedPublicSpans.length > 0).length, publicExactAdmissions: evaluated.filter(value => value.expectedPublicCount > 0 && value.exact).length, privateUnits: cohort.filter(unit => plan.labels.find(label => label.id === unit.id)!.expectedPrivate).length, privateLeakUnits: evaluated.filter(value => value.privateLeak).length, exactAll: evaluated.filter(value => value.exact).length, decisions: evaluated }];
  }))]));
}

export async function executePlan(plan: Plan, units: Unit[], output: string, options: { apiKey: string; fetch?: typeof fetch; now?: () => number }): Promise<unknown> {
  check(plan.model === MODEL_VERSION && plan.privacyThreshold === .10 && JSON.stringify(plan.limits) === JSON.stringify(LIMITS), 'model/threshold/budget changed');
  check(JSON.stringify(plan.sourceHashes) === JSON.stringify(sourceHashes()), 'source/candidate changed after freeze');
  const regenerated = await buildPlan(units, plan.labels, plan.inputHashes);
  check(JSON.stringify(regenerated.calls) === JSON.stringify(plan.calls), 'frozen request hashes changed');
  mkdirSync(output, { recursive: false, mode: 0o700 });
  const log = (event: unknown) => appendFileSync(`${output}/events.jsonl`, `${JSON.stringify(event)}\n`, { mode: 0o600, flush: true });
  const now = options.now ?? Date.now;
  const started = now();
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(new Error('privacy calibration wall deadline')), LIMITS.wallMs);
  const totals = { consumedJev: 0, actualStartedJev: 0, actualStartedExa: 0, requestBytes: 0, responseBytes: 0, measuredInputTokens: 0, usageKnownCalls: 0, usageUnknownCalls: 0 };
  const observations: Observation[] = [];
  let stopReason = 'completed';
  log({ type: 'started', planHash: hash(JSON.stringify(plan)), limits: LIMITS });
  try {
    for (const [sequence, call] of plan.calls.entries()) {
      if (controller.signal.aborted || now() - started >= LIMITS.wallMs) { stopReason = 'wall_budget'; break; }
      if (totals.usageUnknownCalls) { stopReason = 'unknown_usage'; break; }
      if (totals.consumedJev >= LIMITS.maxAttempts || totals.measuredInputTokens + LIMITS.pendingReserve > LIMITS.maxInputTokens || totals.requestBytes + call.requestBytes > LIMITS.maxTotalRequestBytes) { stopReason = 'budget_reserve'; break; }
      if (JSON.stringify(plan.sourceHashes) !== JSON.stringify(sourceHashes())) { stopReason = 'source_freeze_changed'; break; }
      const unit = units.find(value => value.id === call.unitId)!;
      let usage: number | undefined;
      let namedAnswers: Record<string, JevAnswer> | undefined;
      const requestStarted = now();
      const fetcher: typeof fetch = async (url, init) => {
        check(String(url) === 'https://api.typesafe.ai/v1/systemone' && typeof init?.body === 'string' && hash(init.body) === call.bodyHash, 'actual request differs from frozen request');
        const requestBytes = Buffer.byteLength(init.body);
        check(requestBytes === call.requestBytes && requestBytes <= LIMITS.maxRequestBytes, 'actual request byte cap');
        totals.actualStartedJev++; totals.requestBytes += requestBytes;
        log({ type: 'request_sent', sequence, unitId: unit.id, arm: call.arm, requestBytes, bodyHash: call.bodyHash });
        const response = await (options.fetch ?? fetch)(url, init);
        log({ type: 'response_headers', sequence, status: response.status });
        const reader = response.body?.getReader();
        const stream = reader ? new ReadableStream<Uint8Array>({ async pull(target) { const chunk = await reader.read(); if (chunk.done) target.close(); else { totals.responseBytes += chunk.value.byteLength; target.enqueue(chunk.value); } }, async cancel(reason) { await reader.cancel(reason); } }) : null;
        return new Response(stream, { status: response.status, statusText: response.statusText, headers: response.headers });
      };
      const context: ProviderContext = { signal: controller.signal, now: modelNow, consume: provider => { check(provider === 'jev' && totals.consumedJev < LIMITS.maxAttempts, 'provider/attempt budget'); totals.consumedJev++; }, recordUsage: value => { check(typeof value.inputTokens === 'number' && Number.isSafeInteger(value.inputTokens) && value.inputTokens >= 0 && usage === undefined, 'invalid or duplicate usage'); usage = value.inputTokens; totals.measuredInputTokens += value.inputTokens; } };
      const actual = createJevClient({ apiKey: options.apiKey, fetch: fetcher });
      log({ type: 'attempt_started', sequence, unitId: unit.id, arm: call.arm });
      try {
        const admission = await classify(call.arm, { ask: async (request, ctx) => { check(hash(encode(request)) === call.bodyHash, 'classifier request freeze changed'); namedAnswers = await actual.ask(request, ctx); return namedAnswers; } })(input(unit), context);
        // Retain named outputs and exact admitted spans only. No raw prompts or keys.
        const observation = { unitId: unit.id, arm: call.arm, status: 'completed', admission: { ...admission, claims: admission.claims.map(claim => ({ ...claim, originalText: '[control input retained separately]', checkedText: '[exact control span retained separately]' })) }, answers: namedAnswers };
        observations.push(observation); log({ type: 'attempt_terminal', sequence, durationMs: now() - requestStarted, inputTokens: usage ?? null, ...observation });
      } catch (error) {
        const observation = { unitId: unit.id, arm: call.arm, status: 'failed' };
        observations.push(observation); log({ type: 'attempt_terminal', sequence, ...observation, durationMs: now() - requestStarted, inputTokens: usage ?? null, error: error instanceof Error ? error.message : String(error) }); stopReason = 'terminal_failure';
      }
      if (usage === undefined) { totals.usageUnknownCalls++; stopReason = 'unknown_usage'; } else totals.usageKnownCalls++;
      if (totals.measuredInputTokens > LIMITS.maxInputTokens) stopReason = 'measured_token_budget';
      if (stopReason !== 'completed') break;
    }
  } finally { clearTimeout(timeout); }
  const report = { schemaVersion: 1, planHash: hash(JSON.stringify(plan)), stopReason, completedAllCalls: observations.length === plan.calls.length && stopReason === 'completed', durationMs: now() - started, totals, cost: { exaUsd: 0, jevUsd: null, reason: 'Output tokens and Jev pricing unavailable; no invented total cost.' }, metrics: metrics(plan, units, observations), observations };
  log({ type: 'finished', stopReason, totals });
  writeFileSync(`${output}/report.json`, `${JSON.stringify(report, null, 2)}\n`, { flag: 'wx', mode: 0o600 });
  return report;
}

async function main(): Promise<void> {
  const [action, ...args] = process.argv.slice(2);
  if (action === 'prepare') {
    check(args.length === 4, 'prepare requires units labels adjudication new-plan paths');
    const [unitsPath, labelsPath, adjudicationPath, planPath] = args;
    const raw = Object.fromEntries([unitsPath, labelsPath, adjudicationPath].map(path => [resolve(path), readFileSync(path, 'utf8')]));
    const adjudication = JSON.parse(raw[resolve(adjudicationPath)]);
    check(adjudication.frozen === true && adjudication.independentReviewComplete === true, 'independent adjudication must be frozen');
    const plan = await buildPlan(JSON.parse(raw[resolve(unitsPath)]).units, JSON.parse(raw[resolve(labelsPath)]).labels, Object.fromEntries(Object.entries(raw).map(([path, body]) => [path, hash(body)])));
    writeFileSync(planPath, `${JSON.stringify({ ...plan, unitsPath: resolve(unitsPath) }, null, 2)}\n`, { flag: 'wx', mode: 0o600 });
    const saved = JSON.parse(readFileSync(planPath, 'utf8'));
    writeFileSync(`${planPath}.sha256`, `${hash(JSON.stringify(saved))}\n`, { flag: 'wx', mode: 0o600 });
    process.stdout.write(`Prepared ${plan.calls.length} frozen direct-classifier calls; no provider calls.\n`);
  } else if (action === 'execute') {
    check(args.length === 2, 'execute requires frozen-plan/new-output-directory');
    const [planPath, output] = args;
    const plan = JSON.parse(readFileSync(planPath, 'utf8')) as Plan & { unitsPath: string };
    check(hash(JSON.stringify(plan)) === readFileSync(`${planPath}.sha256`, 'utf8').trim(), 'plan hash mismatch');
    for (const [path, expected] of Object.entries(plan.inputHashes)) check(hash(readFileSync(path)) === expected, 'frozen input/labels changed');
    check(process.env.JEV_API_KEY?.trim(), 'JEV_API_KEY required from environment');
    const report = await executePlan(plan, JSON.parse(readFileSync(plan.unitsPath, 'utf8')).units, output, { apiKey: process.env.JEV_API_KEY! }) as { stopReason: string; completedAllCalls: boolean; totals: unknown };
    process.stdout.write(`${JSON.stringify({ stopReason: report.stopReason, completedAllCalls: report.completedAllCalls, totals: report.totals, reportPath: `${output}/report.json` })}\n`);
  } else throw new Error('Explicit prepare or execute required; no default provider action.');
}
if (process.argv[1] && import.meta.url === pathToFileURL(resolve(process.argv[1])).href) main().catch(error => { process.stderr.write(`${error instanceof Error ? error.message : String(error)}\n`); process.exitCode = 1; });
