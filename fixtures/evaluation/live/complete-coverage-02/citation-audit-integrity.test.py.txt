import copy
import importlib.util
from pathlib import Path
import unittest

spec = importlib.util.spec_from_file_location('audit', Path(__file__).with_name('audit-integrity.py'))
audit = importlib.util.module_from_spec(spec)
spec.loader.exec_module(audit)


class AuditIntegrity(unittest.TestCase):
    def setUp(self):
        self.cases = [{'caseId': 'SYNTHETIC-1', 'verdict': 'supported', 'selectedEvidence': [
            {'id': 'a', 'url': 'https://source-a.org/r', 'excerpt': 'Synthetic exact excerpt.'},
            {'id': 'b', 'url': 'https://source-b.org/r', 'excerpt': 'Independent wording; origin not implied.'}]}]
        self.inspections = [dict(caseId='SYNTHETIC-1', evidenceId=source['id'], url=source['url'],
            excerptSha256=audit.digest(source['excerpt']), inspectionMethod='agent', observedAt='2026-11-14T00:00:00Z',
            provenanceNotes='Origin not independently established.', accessStatus='accessible', pageSha256='synthetic-body-hash',
            exactScopeVerified=True, substantiveVerified=True) for source in self.cases[0]['selectedEvidence']]

    def test_all_original_selected_references_required(self):
        self.assertEqual(audit.check_audit(self.cases, self.inspections)['fullyVerifiedCases'], 1)
        self.assertEqual(audit.check_audit(self.cases, self.inspections[:1])['fullyVerifiedCases'], 0)

    def test_inaccessible_partial_or_error_page_prevents_full_credit(self):
        for status in ('inaccessible', 'partial', 'error_page'):
            rows = copy.deepcopy(self.inspections)
            rows[1]['accessStatus'] = status
            self.assertEqual(audit.check_audit(self.cases, rows)['fullyVerifiedCases'], 0)

    def test_missing_scope_or_substance_prevents_full_credit(self):
        for field in ('exactScopeVerified', 'substantiveVerified'):
            rows = copy.deepcopy(self.inspections)
            rows[1][field] = False
            self.assertEqual(audit.check_audit(self.cases, rows)['fullyVerifiedCases'], 0)

    def test_substitutions_and_method_mislabel_are_rejected(self):
        for field, value in [('evidenceId', 'unselected'), ('url', 'https://other.org/'),
                             ('excerptSha256', 'wrong'), ('inspectionMethod', 'human')]:
            rows = copy.deepcopy(self.inspections)
            rows[1][field] = value
            with self.assertRaises(AssertionError):
                audit.check_audit(self.cases, rows)

    def test_duplicate_and_definitive_without_citation_are_rejected(self):
        with self.assertRaises(AssertionError):
            audit.check_audit(self.cases, self.inspections + self.inspections[:1])
        self.cases[0]['selectedEvidence'] = []
        with self.assertRaises(AssertionError):
            audit.check_audit(self.cases, [])

    def test_content_verification_does_not_claim_origin_or_semantic_automation(self):
        result = audit.check_audit(self.cases, self.inspections)
        self.assertFalse(result['independenceProvenByThisCheck'])
        self.assertFalse(result['semanticAssessmentMechanized'])


if __name__ == '__main__':
    unittest.main()
