def sha(value):
    return hashlib.sha256(value).hexdigest()


def read(name):
    return json.loads((DIRECTORY / name).read_bytes())


def encoded(value):
    return (json.dumps(value, indent=2, ensure_ascii=False) + '\n').encode()


def normalize_style(value):
    value = value.get('style') if isinstance(value, dict) else value
    return {'first_person_assertion': 'first_person_public',
            'public_assertion_i_think': 'first_person_public'}.get(value, value)


def first(value, *keys):
    return next((value[key] for key in keys if value.get(key) is not None), None)


def bounded_quote(reference, public_claim):
    """Select a contiguous <=75-word window from the author's inspected excerpts.

    This selects presentation material, not truth labels. Labels and scope notes
    remain those prospectively authored and independently reviewed in each part.
    """
    material = first(reference, 'exactExtractedMaterial', 'supportingTextWindows',
                     'substantivePassages', 'matchedTextWindows')
    passages = [material] if isinstance(material, str) else material
    if not passages:
        raise ValueError('Missing inspected material')
    correct = first(reference, 'correctWorldFact', 'correctRelation',
                    'contraryValueForFalse', 'materialScope', 'exactScopeNotes') or ''
    ignored = set('the a an and or of in to is was were are on by for with from at as it its this that not'.split())
    target = [x.lower() for x in re.findall(r'[A-Za-z]+|\d+', str(correct) + ' ' + public_claim)
              if x.lower() not in ignored]
    windows = []
    for passage in passages:
        words = passage.split()
        for start in range(max(1, len(words) - 74)):
            quote = ' '.join(words[start:start + 75])
            present = set(x.lower() for x in re.findall(r'[A-Za-z]+|\d+', quote))
            score = sum((3 if token.isdigit() else 1) for token in set(target) & present)
            # Prefer substantive relation words over navigation/related questions.
            score -= quote.count('?') * 2
            windows.append((score, -len(windows), quote))
    quote = max(windows)[2]
    assert 0 < len(quote.split()) <= 75
    return quote

        assert reference_gold == gold['verdict'], label

        timestamp = first(reference, 'inspectedAt', 'observedAt')
        status = first(reference, 'httpStatus', 'status')
        assert body_hash and len(body_hash) == 64 and timestamp and status == 200, label
        assert all(text in case['text'] for text in case['allowedPublic'])
        assert gold['acceptedTexts'] == case['allowedPublic']
        if style == 'mixed_private_public':
            assert len(case['allowedPublic']) == 1 and case['text'] not in case['allowedPublic']
        if style == 'first_person_public':
            assert case['text'].startswith('I think ') and case['text'] in case['allowedPublic']
            assert case['text'][8:] in case['allowedPublic']
        public = min(case['allowedPublic'], key=len)
        quote = bounded_quote(reference, public)
        references.append({'label': label, 'expectedVerdict': gold['verdict'],
                           'worldTruth': polarity == 'true', 'url': reference['url'],
                           'finalURL': first(reference, 'finalURL', 'url'),
                           'inspectedAt': timestamp, 'httpStatus': status,
                           'responseBodySHA256': body_hash,
                           'responseBytes': first(reference, 'responseBytes', 'bodyBytes'),
                           'quote': quote, 'quoteWords': len(quote.split()),
                           'quoteSHA256': sha(quote.encode()),

                           'correctRelation': first(reference, 'correctWorldFact', 'correctRelation', 'contraryValueForFalse')

                           'inspectionMethod': first(reference, 'inspectionMethod', 'method'),
                           'sourcePart': name, 'sourceReferenceSHA256': sha(encoded(reference)),
                           'selectionNote': 'Bounded contiguous normalized text from the author-inspected source excerpts; truth/scope labels are not inferred by the assembler.',
                           'runtimeReferenceInjected': False,
                           'underlyingOriginIndependentlyAuthenticated': False})

                           'syntheticAuthored': True, 'sourcePart': name,
                           'independentInput': True, 'containsRealParticipantData': False}

        part_counts[polarity] += 1
        part_counts[style] += 1
