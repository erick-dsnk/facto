function attempt(kind='jev',dispatched=false){return{kind,ordinal:1,dispatched,requestBytes:1000,startedMs:0,durationMs:null,inputTokens:null,unknownInputTokenReservation:kind==='jev'?128000:0,status:null,outcome:'pending',costUSD:null};}
function caseReport(sample:any,attempts:any[],code?:string,wrong=false){const isFact=sample.expected.claims.length===1;return{label:sample.label,status:code?.startsWith('harness_')?'local_stop':'observed',terminalReceived:true,terminalType:code?'pipeline.error':isFact?'claim.result':'claim.skipped',admittedPublic:isFact?sample.allowedPublic:[],rejectedAdmissionCount:0,admission:{action:isFact?'check':'skip'},attempts,localStops:code?.startsWith('harness_')?[code]:[],events:[code?{type:'pipeline.error',reasonCode:code,presentation:{action:'respond',reasonCode:code}}:isFact?{type:'claim.result',claimId:'synthetic-claim',checkedPublicText:sample.text,verdict:wrong?'supported':sample.expected.claims[0].verdict,cached:false,selectedEvidenceIds:[],evidence:[],presentation:{action:'respond',reasonCode:'manual'}}:{type:'claim.skipped',presentation:{action:'respond',reasonCode:'skip'}}]};}
function runnerFactory(config:{code?:string;wrong?:boolean;unknown?:boolean;overrun?:number;lateThrow?:boolean;actualExposure?:boolean;controlOnlyCode?:boolean}={},state={calls:0,wire:0}){
 return Object.assign(async(options:any)=>{
  state.calls++;assert.equal(options.cases.length,1);assert.ok(options.limits.wallMs<=60000);const sample=options.cases[0],attempts:any[]=[];
  const dispatch=async(kind='jev')=>{const a=attempt(kind);if(kind==='search')a.query=config.actualExposure?'SYNTHETIC_PRIVATE_CONTROL_MARKER':sample.text;attempts.push(a);
   try{await options.onProgress({phase:'dispatch_reserved',label:sample.label,attempts,localStops:[],totals:ledger(attempts)});}catch{attempts.pop();return false;}
   a.dispatched=true;state.wire++;a.status=200;a.durationMs=1;a.outcome='response';a.inputTokens=kind==='jev'?(config.unknown?null:config.overrun??17):null;a.unknownInputTokenReservation=kind==='jev'&&config.unknown?128000:0;
   await options.onProgress({phase:'attempt_finished',label:sample.label,attempts,localStops:[],totals:ledger(attempts)});return true;
  };
  let code=config.controlOnlyCode&&sample.expected.claims.length?undefined:config.code;if(!await dispatch())code='harness_checkpoint_failure';
  if(!code&&sample.expected.claims.length&&!config.unknown){await dispatch('search');await dispatch();}
  if(config.actualExposure&&!sample.expected.claims.length)await dispatch('search');
  const report=caseReport(sample,attempts,code,config.wrong);await options.onCase(report);
  if(config.lateThrow)throw new Error('synthetic_unknown_details');
  return{completed:!code?.startsWith('harness_'),transportFailure:null,counts:{providerErrors:code?1:0},totals:ledger(attempts),cases:[report],predeclaredCases:[]};
 },{state});
}
