function inspect(kind: string, variant: string, request: JevRequest, details: Record<string, unknown> = {}) {
  const bytes = Buffer.byteLength(JSON.stringify({ model: MODEL_VERSION, state: request.state, questions: request.questions }));
  maximumBytes = Math.max(maximumBytes, bytes);
  observations.push({ kind, variant, requestBytes: bytes, questions: Object.keys(request.questions).length, ...details });
}

}
