async function readCaptured(file:string) {
  const pin=savedPins.get(file);assert.ok(pin,'capture file missing from frozen manifest:'+file);
  const bytes=await readFile(file);assert.equal(bytes.length,pin.bytes);assert.equal(hash(bytes),pin.sha256);
  return {decoded:JSON.parse(bytes.toString('utf8')),sha256:pin.sha256,bytes:pin.bytes};
}
function run(parser:typeof old,response:unknown,requestedURLs:string[]|null) {
  try {
    const value=requestedURLs ? parser.parseExaContentsResponse(response,requestedURLs) : {pages:parser.parseExaResponse(response),pageFailures:[]};
    return {kind:'parsed' as const,pages:value.pages,pageFailures:value.pageFailures};
  } catch(error) {
    const e=error as {name?:string;code?:string;retryable?:boolean};
    return {kind:'error' as const,name:e.name??null,code:e.code??null,retryable:e.retryable??null};
  }
}
function pageSummary(page:any) {
  return {url:page.url,title:page.title??null,publishedDate:page.publishedDate??null,publisher:page.publisher??null,
    textPresent:typeof page.text==='string',textCodepoints:typeof page.text==='string'?Array.from(page.text).length:0,
    textSHA256:typeof page.text==='string'?hash(page.text):null,
    highlightCount:Array.isArray(page.highlights)?page.highlights.length:0,
    highlightSHA256:Array.isArray(page.highlights)?page.highlights.map((h:string)=>hash(h)):[],
    entireNormalizedPageSHA256:hash(canonical(page))};
}
function summary(result:ReturnType<typeof run>) {
  return result.kind==='parsed'?{kind:result.kind,pages:result.pages.map(pageSummary),pageFailures:result.pageFailures,

}
