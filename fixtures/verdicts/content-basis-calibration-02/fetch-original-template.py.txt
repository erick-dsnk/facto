from pathlib import Path
import json,hashlib,datetime,urllib.request,urllib.error,concurrent.futures,re
from html.parser import HTMLParser
base=Path('/workspace/task-evidence');out=base/'automatic08-citation-audit'/'origin-fetches';packet=json.loads((base/'automatic08-selected-citations.json').read_text())
urls=list(dict.fromkeys(c['evidence']['url']for c in packet['citations']))
class TextParser(HTMLParser):
 def __init__(self):super().__init__();self.skip=0;self.parts=[]
 def handle_starttag(self,t,a):
  if t in ('script','style','noscript'):self.skip+=1
 def handle_endtag(self,t):
  if t in ('script','style','noscript'):self.skip=max(0,self.skip-1)
 def handle_data(self,d):
  if not self.skip:self.parts.append(d)
def fetch(url):
 id=hashlib.sha256(url.encode()).hexdigest()[:16];record={'requestedUrl':url,'fetchedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),'freshRequest':True,'method':'GET','requestOrdinal':urls.index(url)+1}
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0 (independent public citation audit)'}),timeout=30) as r:
   raw=r.read(2000001);record.update(status=r.status,finalUrl=r.url,contentType=r.headers.get('Content-Type'),truncated=len(raw)>2000000);raw=raw[:2000000]
 except urllib.error.HTTPError as e:
  raw=e.read(2000000);record.update(status=e.code,finalUrl=e.url,contentType=e.headers.get('Content-Type'),error=str(e),truncated=False)
 except Exception as e:raw=b'';record.update(status=None,error=str(e),truncated=False)
 record.update(bytes=len(raw),bodySHA256=hashlib.sha256(raw).hexdigest(),bodyFile=str(out/(id+'.body')))
 (out/(id+'.body')).write_bytes(raw);html=raw.decode('utf-8',errors='replace');parser=TextParser();parser.feed(html);text=' '.join(' '.join(parser.parts).split());(out/(id+'.text.txt')).write_text(text+'\n');title=re.search(r'<title[^>]*>(.*?)</title>',html,re.S|re.I);record.update(title=re.sub(r'\s+',' ',title[1])if title else None,textFile=str(out/(id+'.text.txt')),textSHA256=hashlib.sha256((text+'\n').encode()).hexdigest());(out/(id+'.json')).write_text(json.dumps(record,indent=2)+'\n');return record
with concurrent.futures.ThreadPoolExecutor(max_workers=6)as pool:records=list(pool.map(fetch,urls))
(out/'fetch-manifest.json').write_text(json.dumps({'freshPublicHttpRequests':6,'records':records},indent=2)+'\n')
for r in records:print(r['requestedUrl'],r.get('status'),r.get('title'),r['bytes'])
