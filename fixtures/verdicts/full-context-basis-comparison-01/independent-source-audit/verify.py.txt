B=pathlib.Path(__file__).resolve().parent

manifest=json.loads((B/'MANIFEST.json').read_text())
for f in manifest['files']:assert H(B/f['path'])==f['sha256'],f['path']

 assert m['httpStatus']==200 and not m['bodyTruncated']
 assert H(root/m['responseBodyFile'])==r['bodySHA256']==m['responseBodySHA256']
 assert H(root/m['extractedTextFile'])==r['textSHA256']==m['extractedTextSHA256']

for r in reuse:
