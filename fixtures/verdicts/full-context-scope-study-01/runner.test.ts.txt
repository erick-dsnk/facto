function fakeFetch(tokens: number | undefined = 100): typeof fetch {
    return async (_url, init) => {
        const req = JSON.parse(String(init!.body));
        const answers = Object.fromEntries(Object.entries(req.questions).map(([name, raw]) => {
            const q = raw as {
                type: string;
                instructions: string;
                criteria?: Record<string, string>;
            };
            if (q.type === 'noul') {
                let noul = name.endsWith('_context_established') ? 0 : .94;
                const p = /^e(\d+)_scope$/.exec(name);
                if (p && q.instructions.includes('This asks coverage, not agreement or a verdict.')) {
                    const id = req.state.evidence[Number(p[1])].id;
                    const n = Number(/^source-(\d+)-/.exec(id)![1]);
                    noul = n % 2 === 0 ? .96 : .1;
                }
                return [name, { noul }];
            }
            const choice = name === 'verdict' ? 'supported' : name === 'selected_evidence' ? 'e0' : name === 'claim_risk' ? 'ordinary' : name.endsWith('_stance') ? 'supports' : name.endsWith('_origin') ? 'distinct_origins' : name.endsWith('_content_basis') ? 'original_record' : 'none';
            return [name, { choice, probabilities: Object.fromEntries(Object.keys(q.criteria!).map(k => [k, k === choice ? 1 : 0])) }];
        }));
        return new Response(JSON.stringify({ answers, ...(tokens === undefined ? {} : { usage: { input_tokens: tokens } }) }), { status: 200 });
    };
}
test('only scope instructions change; exact full state, source count and every other question remain identical', async () => {
    const a = await captureRequest(contexts[0]!, 'original_full', 'baseline'), b = await captureRequest(contexts[0]!, 'original_full', 'scope_candidate');
    assert.deepEqual(a.state, b.state);
    assert.deepEqual(Object.keys(a.questions), Object.keys(b.questions));
    for (const [name, q] of Object.entries(a.questions)) {
        if (/^e\d+_scope$/.test(name)) {
            assert.equal(b.questions[name]!.type, 'noul');
            assert(b.questions[name]!.instructions.startsWith(q.instructions + ' '));
        }
        else
            assert.deepEqual(b.questions[name], q);
    }
    assert(!JSON.stringify(b).includes('expectedScope'));
    assert(!JSON.stringify(b).includes('labelReason'));
    assert(!JSON.stringify(b).includes('questionKey'));
});
test('48 requests keep separate balanced strata, all companions, one prospective order and both existing orders', async () => {
    const p = await plan();
    assert.equal(p.calls.length, 48);
    assert.deepEqual(p.deferredCohorts, ['scope16_live_rerun']);
    for (const stratum of ['prospective', 'existing_safety_diagnostic']) {
        const calls = p.calls.filter(c => c.stratum === stratum);
        assert.equal(calls.length, 24);
        const pairs = Array.from({ length: 12 }, (_, n) => calls.slice(n * 2, n * 2 + 2));
        assert.equal(pairs.filter(x => x[0]!.arm === 'baseline').length, 6);
        assert.equal(pairs.filter(x => x[0]!.arm === 'scope_candidate').length, 6);
        for (const pair of pairs) {
            assert.equal(pair[0]!.caseId, pair[1]!.caseId);
            assert.equal(pair[0]!.condition, pair[1]!.condition);
            assert.deepEqual(pair[0]!.evidenceOrder, pair[1]!.evidenceOrder);
            assert.notEqual(pair[0]!.arm, pair[1]!.arm);
        }
    }
    assert(p.calls.filter(c => c.stratum === 'prospective').every(c => c.condition === 'original_full'));
    for (const c of contexts) {
        const calls = p.calls.filter(x => x.caseId === c.caseId);
        assert.equal(calls.length, c.stratum === 'prospective' ? 2 : 4);
        for (const call of calls)
            assert.deepEqual([...call.evidenceOrder].sort(), c.records.map(x => x.id).sort());
    }
});

test('wall abort settles despite an abort-ignoring reader and never-settling cancellation', async () => {
    const frozen = await plan(), realSetTimeout = globalThis.setTimeout;
    let canceled = false;
    globalThis.setTimeout = ((callback: any, delay?: number, ...args: any[]) => realSetTimeout(callback, delay === LIMITS.wallMs ? 1 : delay, ...args)) as typeof setTimeout;
    try {
        const r: any = await Promise.race([executePlan(frozen, contexts, output(), { apiKey: 'offline-fake-key', now: () => 0, fetch: async () => new Response(new ReadableStream({ pull: () => new Promise(() => { }), cancel: () => { canceled = true; return new Promise(() => { }); } }), { status: 200 }) }), new Promise((_, reject) => realSetTimeout(() => reject(Error('runner failed to settle')), 150))]);
        assert.equal(r.stopReason, 'terminal_failure');
        assert.equal(r.primaryFailure.reason, 'absolute_wall_deadline');
        assert.equal(r.unknownUsage, true);
        assert.equal(canceled, true);
    }
    finally {
        globalThis.setTimeout = realSetTimeout;
    }
});
test('late abort-ignoring success cannot mutate finalized totals, answers, observations or report', async () => {
    const frozen = await plan(), realSetTimeout = globalThis.setTimeout;
    let resolveLate!: (response: Response) => void;
    let capturedUrl: any, capturedInit: RequestInit | undefined;
    globalThis.setTimeout = ((callback: any, delay?: number, ...args: any[]) => realSetTimeout(callback, delay === LIMITS.wallMs ? 1 : delay, ...args)) as typeof setTimeout;
    try {
        const dir = output();
        const r: any = await Promise.race([executePlan(frozen, contexts, dir, { apiKey: 'offline-fake-key', now: () => 0, fetch: async (url, init) => { capturedUrl = url; capturedInit = init; return new Promise(resolve => resolveLate = resolve); } }), new Promise((_, reject) => realSetTimeout(() => reject(Error('runner failed to settle')), 150))]);
        const returned = JSON.stringify(r), saved = readFileSync(join(dir, 'report.json'), 'utf8');
        resolveLate(await fakeFetch(9999)(capturedUrl, capturedInit));
        await new Promise(resolve => realSetTimeout(resolve, 10));
        assert.equal(JSON.stringify(r), returned);
        assert.equal(readFileSync(join(dir, 'report.json'), 'utf8'), saved);
        assert.equal(r.totals.measuredInputTokens, 0);
        assert.equal(r.observations[0].answers, undefined);
        assert.equal(r.observations[0].policyDecision, undefined);
    }
    finally {
        globalThis.setTimeout = realSetTimeout;
    }
});
