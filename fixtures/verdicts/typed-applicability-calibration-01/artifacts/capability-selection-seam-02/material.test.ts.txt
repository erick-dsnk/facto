const claim = 'The Meridian bridge opened in 1889.';
const source = (id: string, excerpt: string): Evidence => ({ id, url: `https://${id}.org/record`, title: 'Bridge register',
 publisher: null, publishedAt: '2026-11-14', retrievedAt: '2026-11-14T10:00:00Z', contentBasis: 'highlight', excerpt,
 sourceGroup: `${id}.org`, echoGroup: null, sourceRole: 'unknown', directlyAddressesClaim: false,
 matchesTimePlaceAndPopulation: false, stance: 'unrelated', stanceProbability: null });
const judgment = (records: Evidence[]): AssessedJudgment => ({ checkedText: claim, label: 'insufficient_evidence',
 probabilities: { supported: 0, contradicted: 0, misleading: 0, insufficient_evidence: 1 },
 evidence: records.map(record => ({ ...record, directlyAddressesClaim: true, matchesTimePlaceAndPopulation: true, stance: 'supports', stanceProbability: .99 })),
 selectedEvidenceIds: [], misleadingReason: null,
 sourceAssessments: Object.fromEntries(records.map(record => [record.id, { contentBasis: 'original_record', confidence: .99 }])),
 claimRisk: { level: 'ordinary', confidence: .99 }, originAssessments: [] });
function simple() {
 const old = { ...source('alpha', `Register M-17. ${claim} However, public access began in 1890; the earlier date describes a private ceremony. Keeper: Municipal Archive.`), contentBasis: 'page_text' as const };
 const next = { ...old, contentBasis: 'highlight' as const, excerpt: claim, retrievedAt: '2026-11-14T10:00:05Z' };
 const handle = captureSelectionMaterial({ existing: [old], incoming: [next], checkedText: claim, priorJudgment: judgment([old]) });
 return { old, next, handle, snapshot: inspectMaterial(handle) };
}
function lifecycle() {
 let clock = Date.parse('2026-11-14T10:00:00Z');
 const session = new Session(() => clock);
 const record: RequestRecord = { input: { schemaVersion: 1, sessionId: session.id, sessionEpoch: session.epoch, requestId: 'offline-request', revision: 1,
  text: claim, context: [], mode: 'typed', language: 'en', capturedAt: new Date(clock).toISOString(),
  capture: { source: 'typed_fixture', segmentId: 'offline-segment', confidence: 1 }, userInitiated: true },
  budget: new AttemptBudget(), controller: new AbortController(), acceptedAt: clock,
  acceptedMonotonic: 0, state: 'running', continuationCount: 0, results: new Map(), fingerprint: 'synthetic' };
 session.requests.set(record.input.requestId, record);
 const { handle, snapshot } = simple();
 let generation = snapshot.materialGeneration, installation = 'installed-generation-1';
 const admitted: AcceptedClaim = { id: 'offline-request-claim-0', originalText: claim, checkedText: claim,
  sourceSpan: { start: 0, end: Array.from(claim).length }, admission: { assertionStatus: 'asserted', hasPracticalConsequence: null,
   matchedSelectedTopic: null, admissionAction: 'check', reasonCode: 'synthetic-admitted-public-claim' } };
 const owner = captureRuntimeOwner(handle, { session, record, claimSlot: 0, admittedClaim: admitted,
  readMaterialGeneration: () => generation, readInstallGeneration: () => installation });
 return { session, record, owner, handle, snapshot, admitted, advance: (ms: number) => { clock += ms; },
  replaceMaterial: () => { generation = 'changed'; }, retireInstall: () => { installation = 'retired'; } };
}

for (const variant of ['cancel', 'epoch', 'replace', 'session_expiry', 'deadline', 'material', 'install', 'done'] as const) {
 test(`late completion under ${variant} cannot cross owner gate or populate a cache`, async () => {
  const l = lifecycle(); let resolve!: (v: string) => void; let commits = 0;
  const pending = awaitCurrentMaterial(l.owner, () => new Promise<string>(r => { resolve = r; })).then(() => { commits++; });
  const rejected = assert.rejects(pending);
  if (variant === 'cancel') l.session.cancel(l.record.input.requestId);
  if (variant === 'epoch') { l.session.pause(); l.session.active = true; }
  if (variant === 'replace') l.session.requests.set(l.record.input.requestId, { ...l.record, input: { ...l.record.input, revision: 2 }, controller: new AbortController() });
  if (variant === 'session_expiry') l.session.expiresAt = l.session.now();
  if (variant === 'deadline') l.advance(10000);
  if (variant === 'material') l.replaceMaterial();
  if (variant === 'install') l.retireInstall();
  if (variant === 'done') l.record.state = 'done';
  resolve('late valid-looking payload'); await rejected; assert.equal(commits, 0);
  assert.equal(l.record.budget.jev, 0); assert.equal(l.session.cache.size, 0); assert.equal(l.record.results.size, 0);
 });
}
