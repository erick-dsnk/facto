function validity(s: Awaited<ReturnType<typeof setup>>, r: any) {

function inventory(value: unknown, path: string, rows: unknown[], seen = new WeakSet<object>()) {
 if (!value || (typeof value !== 'object' && typeof value !== 'function') || seen.has(value as object)) return;
 seen.add(value as object);
 const proto = Object.getPrototypeOf(value);
 const native = ![Object.prototype, Array.prototype, null].includes(proto);
 rows.push({ path, frozen: Object.isFrozen(value), kind: typeof value === 'function' ? 'callable' : native ? 'native_opaque' : Array.isArray(value) ? 'array' : 'record' });
 if (native || typeof value === 'function') return;
 for (const [key, d] of Object.entries(Object.getOwnPropertyDescriptors(value))) if ('value' in d) inventory(d.value, `${path}.${key}`, rows, seen);
}

 test(`${version}: actual exposed signal method mutation delays prompt cancel, while private owner still prevents authority`, async () => {
  const control = await setup({ adapterDir: directory(version), blockFetch: true });
  let controlSettled = false;
  const controlPending = control.execute().then(() => { controlSettled = true; }, () => { controlSettled = true; });
  await control.started; control.session.cancel(control.record.input.requestId); await pause(30);
  assert.equal(controlSettled, true); control.releaseTransport(); await controlPending;

  const s = await setup({ adapterDir: directory(version), blockFetch: true });
  const signal = s.inspectDispatchOwner(s.dispatchOwner).context.signal as AbortSignal;
  const original = signal.addEventListener;
  assert.equal(Reflect.set(signal, 'addEventListener', () => {}), true);
  let settled = false; let succeeded = false;
  const pending = s.execute().then(() => { settled = true; succeeded = true; }, () => { settled = true; });
  await s.started; s.session.cancel(s.record.input.requestId); await pause(30);
  const promptSettled = settled;
  Reflect.deleteProperty(signal, 'addEventListener'); assert.equal(signal.addEventListener, original);
  s.releaseTransport(); await pending;
  assert.equal(promptSettled, false); assert.equal(succeeded, false); assert.equal(s.record.budget.jev, 1);
  assert.equal(s.session.cache.size, 0); assert.equal(s.record.results.size, 0);
  observations.push({ version, surface: 'exposed_transport_signal', actualMutation: 'shadow addEventListener on returned native object',
    controlSettledAfterCancel: true, mutatedSettledAfter30ms: promptSettled, lateAuthorityIssued: succeeded,
    mockAttemptsCharged: s.record.budget.jev, projectProviderCalls: 0,
    scope: 'Trusted in-process caller can alter exposed native signal; no remote source/provider control demonstrated.' });
 });

async function prototype() {

 const f = resolveFrozenFrame(manifest, 'RC08'), frame = loadFrozenClaimFrame(manifest, 'RC08', f.exactText);
 const evidence = { id: 'target', url: receipt.canonicalURL, excerpt: source.text.slice(receipt.units[0].start, receipt.units[0].end) };
 const context = fixtureContext(f.exactText, [fixtureRecord(evidence), fixtureRecord({ id: 'companion', url: 'https://companion.example.org/record', excerpt: 'Synthetic companion record.' })]);
 context.originLinks = [{ left: context.originNodes[0].key, right: context.originNodes[1].key, relationship: 'unclear', state: { note: 'synthetic origin control' } }];

}
