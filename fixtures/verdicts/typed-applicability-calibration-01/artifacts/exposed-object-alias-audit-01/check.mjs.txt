import { readFile, writeFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { spawnSync } from 'node:child_process';
const root = '/workspace/task-evidence/exposed-object-alias-audit-01';
const hash = b => createHash('sha256').update(b).digest('hex');
const pins = [], packets = [];
for (const name of ['typed-applicability-adapter-05', 'typed-applicability-adapter-06', 'record-capability-prototype-02']) {
 const dir = '/workspace/task-evidence/' + name;
 const raw = await readFile(dir + '/manifest.json'), manifest = JSON.parse(raw);
 for (const [file, value] of Object.entries(manifest.files)) {
  const data = await readFile(dir + '/' + file), expected = typeof value === 'string' ? value : value.sha256;
  if (hash(data) !== expected) throw Error('frozen_packet_hash_mismatch');
  pins.push({ path: dir + '/' + file, sha256: expected });
 }
 packets.push({ path: dir, manifestSHA256: hash(raw), listedFilesVerified: true });
 pins.push({ path: dir + '/manifest.json', sha256: hash(raw) });
}
for (const path of ['audit.test.ts', 'fixture.ts', 'tsconfig.json', 'package.json']) pins.push({ path: `${root}/${path}`, sha256: hash(await readFile(`${root}/${path}`)) });
const runs = [];
for (const [name, executable, args, log] of [
 ['typecheck', '/workspace/stack-20/node_modules/.bin/tsc', ['-p', root + '/tsconfig.json'], 'typecheck.log'],
 ['tests', '/workspace/stack-20/node_modules/.bin/tsx', ['--test', '--test-reporter=tap', root + '/audit.test.ts'], 'tests.tap']]) {
 const result = spawnSync(executable, args, { encoding: 'utf8', timeout: 60000, maxBuffer: 1024 * 1024 });
 await writeFile(root + '/' + log, (result.stdout ?? '') + (result.stderr ?? ''));
 runs.push({ name, exitCode: result.status, error: result.error?.name ?? null, log });
}
const tap = await readFile(root + '/tests.tap', 'utf8');
if (!/^# pass 18$/m.test(tap) || !/^# fail 0$/m.test(tap) || runs.some(r => r.exitCode !== 0)) throw Error('audit_check_failed');
for (const pin of pins) if (hash(await readFile(pin.path)) !== pin.sha256) throw Error('source_or_packet_changed');
await writeFile(root + '/validation.json', JSON.stringify({ kind: 'actual_exposed_object_alias_audit_validation',
 projectProviderCalls: 0, networkCalls: 0, injectedFetchOnly: true, testsPassed: 18, runs,
 frozenPacketsPreserved: packets, sourcePins: pins, known05ContextAliasRepeated: false,
 sourceAndPacketBytesStable: true }, null, 2) + '\n');
console.log(JSON.stringify({ testsPassed: 18, typecheck: 'passed', packetsPreserved: true, projectProviderCalls: 0 }));
