import { readFile, writeFile, mkdir, lstat, realpath } from 'node:fs/promises';
import path from 'node:path';
import { pathToFileURL } from 'node:url';
import { canonical, digest, fail, parseInstalledData, parseReceipt, type Frame, type InstalledData, type Proof, type Receipt } from './types.js';
export const CORPUS_ROOT = '/workspace/task-evidence/record-capability-design';
export const CORPUS_MANIFEST_SHA256 = '61d06043365be55ad91b8f85e01d336b82c481d83585fdcba4906604471cda4e';
export const INSTALL_ROOT = '/workspace/task-evidence/record-capability-prototype-02/installed';
const safeRelative = (p: string) => {
    if (path.isAbsolute(p) || p.split('/').some(x => !x || x === '.' || x === '..') || p.includes('\\'))
        return fail('unsafe_corpus_path');
    return p;
};
/** Deterministic data conversion from one immutable, independently reviewed corpus. No labels are parsed. */
export async function installCorpus(destination = INSTALL_ROOT): Promise<string> {
    const manifestBytes = await readFile(path.join(CORPUS_ROOT, 'manifest.json'));
    if (digest(manifestBytes) !== CORPUS_MANIFEST_SHA256)
        return fail('corpus_manifest_changed');
    const original = JSON.parse(manifestBytes.toString('utf8')) as {
        files: Record<string, {
            sha256: string;
            bytes: number;
        }>;
    };
    const verified = new Map<string, Buffer>();
    for (const [relative, metadata] of Object.entries(original.files)) {
        safeRelative(relative);
        const file = path.join(CORPUS_ROOT, relative);
        if (await realpath(file) !== file || (await lstat(file)).isSymbolicLink())
            return fail('corpus_symlink');
        if (metadata.bytes > 2097152)
            return fail('corpus_file_limit');
        const bytes = await readFile(file);
        if (bytes.length !== metadata.bytes || digest(bytes) !== metadata.sha256)
            return fail('corpus_file_changed');
        verified.set(relative, bytes);
    }
    const read = (relative: string) => {
        const bytes = verified.get(relative);
        if (!bytes)
            return fail('unreviewed_corpus_file');
        return bytes;
    };
    const packet = JSON.parse(read('annotation-inputs.json').toString('utf8')) as any;
    const sourceBindings = JSON.parse(read('reviewed-receipt-bindings.json').toString('utf8')) as any;
    const proofData: Proof[] = packet.proofs.map((p: any) => {
        const r = p.retrieval;
        if (r.status !== 200 || r.truncated)
            return fail('invalid_original_proof');
        return { id: p.id, requestedURL: r.requestedUrl, finalURL: r.finalUrl, redirects: r.redirects?.map((h: any) => h.to) ?? [], acquiredAt: r.completedAt, httpStatus: 200, complete: true, rawBodySHA256: r.bodySHA256, extractionSHA256: r.textSHA256 };
    });
    const artifacts = new Map<string, Buffer>();
    for (const p of packet.proofs) {
        for (const [fileKey, digestKey] of [['bodyFile', 'bodySHA256'], ['textFile', 'textSHA256']] as const) {
            const absolute = p.retrieval[fileKey];
            if (typeof absolute !== 'string' || !absolute.startsWith(CORPUS_ROOT + '/'))
                return fail('external_proof_path');
            const bytes = read(safeRelative(absolute.slice(CORPUS_ROOT.length + 1)));
            if (digest(bytes) !== p.retrieval[digestKey])
                return fail('original_proof_hash');
            artifacts.set(digest(bytes), bytes);
        }
    }
    const receipts: {
        alias: string;
        sha256: string;
        bytes: Buffer;
    }[] = [];
    for (const r of packet.attestations) {
        const sourceReceiptSHA256 = digest(canonical(r));
        const binding = sourceBindings.bindings.find((x: any) => x.corpusAlias === r.attestationId);
        if (binding?.receiptSHA256 !== sourceReceiptSHA256)
            return fail('reviewed_receipt_binding_mismatch');
        const text = artifacts.get(r.extractionSHA256)!.toString('utf8');
        const units = r.recordUnits.map((u: any) => {
            if (u.offsetUnit !== 'unicode_code_points')
                return fail('unknown_original_offset_unit');
            const points = Array.from(text);
            const start = points.slice(0, u.extractionStart).join('').length, end = points.slice(0, u.extractionEnd).join('').length;
            const body = text.slice(start, end);
            if (body !== u.content || digest(body) !== u.contentSHA256)
                return fail('offset_conversion_mismatch');
            return { locator: u.locator, start, end, contentSHA256: u.contentSHA256, materialSpans: [{ role: 'entire_audited_record_unit', start: 0, end: body.length }] };
        });
        const receipt: Receipt = { schemaVersion: 1, sourceReceiptSHA256, issuerId: r.issuer.id, issuerProofIds: [r.issuer.identityAndCompetenceProofRef, ...r.issuer.additionalIdentityProofRefs], canonicalDocumentId: r.document.canonicalId, canonicalURL: r.document.canonicalUrl, versionId: r.document.version, documentClass: r.document.documentClass, relation: r.capability.relationClasses[0], scope: r.document.versionScope, proofId: r.retrievalProofRef, rawBodySHA256: r.bodySHA256, extractionSHA256: r.extractionSHA256, units, originFamily: r.originFamily, temporalMode: r.document.documentClass === 'normative_definition' ? 'immutable_version' : 'historical_release', materialSpanPolicy: 'entire_reviewed_unit_required', limitations: [...r.limitations, ...(r.issuer.identityLimitations ? [r.issuer.identityLimitations] : []), 'Receipt authority derives from the frozen independent corpus review, not from text labels or a hostname.'] };
        parseReceipt(receipt);
        const bytes = Buffer.from(canonical(receipt) + '\n');
        receipts.push({ alias: r.attestationId, sha256: digest(bytes), bytes });
    }
    const frames: Frame[] = packet.units.map((u: any) => { const scope = u.requestedScope; const current = Object.values(scope).some(v => v === 'latest' || v === 'current'); return { id: u.id, exactText: u.proposition, exactTextSHA256: digest(u.proposition), relation: u.requestedRelationClass, materialContext: scope, timeMeaning: current ? 'current' : typeof scope.version === 'string' ? 'explicit_version' : 'unresolved' }; });
    const data: InstalledData = { schemaVersion: 1, corpusManifestSHA256: CORPUS_MANIFEST_SHA256, productionAuthorized: false, receipts: receipts.map(({ alias, sha256 }) => ({ alias, sha256 })), proofs: proofData, frames, revokedReceipts: [] };
    parseInstalledData(data);
    await mkdir(destination, { recursive: false });
    await mkdir(path.join(destination, 'artifacts'));
    await mkdir(path.join(destination, 'receipts'));
    for (const [hash, bytes] of artifacts)
        await writeFile(path.join(destination, 'artifacts', hash), bytes, { flag: 'wx' });
    for (const r of receipts)
        await writeFile(path.join(destination, 'receipts', r.sha256 + '.json'), r.bytes, { flag: 'wx' });
    const installedBytes = Buffer.from(canonical(data) + '\n');
    await writeFile(path.join(destination, 'manifest.json'), installedBytes, { flag: 'wx' });
    return digest(installedBytes);
}
if (import.meta.url === pathToFileURL(process.argv[1] ?? '').href) {
    const hash = await installCorpus();
    await writeFile('/workspace/task-evidence/record-capability-prototype-02/installed-manifest.sha256', hash + '\n');
    console.log(JSON.stringify({ installedManifestSHA256: hash, sourceCorpusManifestSHA256: CORPUS_MANIFEST_SHA256 }));
}
