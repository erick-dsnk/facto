async function input(id = 'RC08', installation = manifest): Promise<EvaluateInput> {
    const u = packet.units.find((x: any) => x.id === id);
    const receipt = resolveReceipt(installation, u.receiptId).receipt;
    return { manifest: installation, artifact: await loadSnapshot(installation, receipt.proofId), receiptAlias: u.receiptId, evidence: { id: u.id, url: receipt.canonicalURL, excerpt: u.excerpt }, frame: loadFrozenClaimFrame(installation, u.id, u.proposition), exactClaimText: u.proposition, context: fixtureContext(u.proposition, [fixtureRecord({ id: u.id, url: receipt.canonicalURL, excerpt: u.excerpt })]), now };
}
function withEvidence(i: EvaluateInput, evidence: typeof i.evidence): EvaluateInput { return { ...i, evidence, context: replaceFixtureTarget(i.context, evidence) }; }
function unavailable(value: unknown, reason?: string) {
    const result = evaluate(value);
    assert.equal(result.kind, 'unavailable');
    if (result.kind === 'unavailable' && reason)
        assert.equal(result.reason, reason);
    return result;
}
function capability(value: unknown) {
    const result = evaluate(value);
    assert.equal(result.kind, 'capability', JSON.stringify(result));
    if (result.kind !== 'capability')
        throw Error('required capability');
    return result;
}
async function temporary(action: (root: string) => Promise<void>) {
    const base = await mkdtemp(path.join(os.tmpdir(), 'record-capability-'));
    const root = path.join(base, 'installed');
    await cp(INSTALL_ROOT, root, { recursive: true });
    try {
        await action(root);
    }
    finally {
        await rm(base, { recursive: true, force: true });
    }
}
async function mutateManifest(root: string, mutate: (data: any) => void) { const file = root + '/manifest.json'; const data = JSON.parse(await readFile(file, 'utf8')); mutate(data); const bytes = canonical(data) + '\n'; await writeFile(file, bytes); return digest(bytes); }
async function mutateReceipt(root: string, alias: string, mutate: (data: any) => void) { const data = JSON.parse(await readFile(root + '/manifest.json', 'utf8')); const entry = data.receipts.find((r: any) => r.alias === alias); const receipt = JSON.parse(await readFile(root + '/receipts/' + entry.sha256 + '.json', 'utf8')); mutate(receipt); const bytes = canonical(receipt) + '\n'; entry.sha256 = digest(bytes); await writeFile(root + '/receipts/' + entry.sha256 + '.json', bytes); const manifestBytes = canonical(data) + '\n'; await writeFile(root + '/manifest.json', manifestBytes); return digest(manifestBytes); }
