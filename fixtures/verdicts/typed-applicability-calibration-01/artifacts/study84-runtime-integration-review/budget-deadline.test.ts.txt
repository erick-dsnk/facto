// Offline reviewer harness. All transport calls are injected; global network is forbidden.
import {test} from 'node:test';
import assert from 'node:assert/strict';
import fs from 'node:fs';
import {syncBuiltinESMExports} from 'node:module';
import {createHash} from 'node:crypto';
import {join} from 'node:path';
import {Session} from '/workspace/stack-20/services/api/src/runtime/session.js';
import {runStudy,assertSources,validatePlan,type Plan} from '/workspace/task-evidence/typed-applicability-calibration-01/runner.js';
const root='/workspace/task-evidence/study84-runtime-integration-review/additional-budget-deadline-checks';
const planPath='/workspace/task-evidence/typed-applicability-calibration-01/applicability42-plan.json';
const sha=(v:string|Uint8Array)=>createHash('sha256').update(v).digest('hex');
globalThis.fetch=async()=>{throw Error('offline_global_fetch_forbidden');};
function plan():Plan{const p=JSON.parse(fs.readFileSync(planPath,'utf8'));validatePlan(p);assertSources(p);return p;}
function response(init?:RequestInit,mode:'valid'|'missing_usage'|'malformed'='valid'){
 const request=JSON.parse(String(init?.body));
 const answers=Object.fromEntries(Object.entries(request.questions).map(([key,value])=>{const q=value as any;if(q.type==='noul')return[key,{noul:.5}];const keys=Object.keys(q.criteria);const choice=key==='claim_risk'?'unknown':key==='selected_evidence'?'none':key==='verdict'?'insufficient_evidence':key.endsWith('_applicability')?'uncertain':keys[keys.length-1]!;return[key,{choice,probabilities:Object.fromEntries(keys.map(k=>[k,k===choice?.99:.01/(keys.length-1)]))}];}));
 if(mode==='malformed')delete answers.verdict;
 return Response.json({answers,...(mode==='missing_usage'?{}:{usage:{input_tokens:10}})});
}
function output(name:string){const path=join(root,name);assert.equal(fs.existsSync(path),false,`fresh reviewer output required: ${path}`);return path;}
const options=(fetch:typeof globalThis.fetch)=>({apiKey:'OFFLINE_INJECTED',fetch,transportMode:'offline-mock' as const});
const journal=(out:string)=>fs.readFileSync(join(out,'checkpoints.jsonl'),'utf8').trim().split('\n').map(x=>JSON.parse(x));


test('actual runner records complete measured block overrun and suppresses all later dispatches',async()=>{
 const out=output('block-overrun');let calls=0;const result=await runStudy(plan(),out,options(async(_url,init)=>{calls++;const body=await response(init).json();body.usage.input_tokens=320001;return Response.json(body);}));
 assert.equal(calls,1);assert.equal(result.rows.length,0);assert.equal(result.stop,'measured_input_token_cap');
 assert.equal(result.guard.totals.measuredInputTokens,320001);assert.equal(result.guard.totals.usageUnknownCalls,0);
 assert.equal(result.guard.perBlock.block1.measuredInputTokens,320001);assert.equal(result.guard.missingCallIds.length,84);
 console.log(JSON.stringify({scenario:'block-overrun',calls,usage:result.guard.totals.measuredInputTokens,stop:result.stop}));
});
for(const advance of [10000,120000])test(`actual runner halts on inherited absolute deadline after clock advances ${advance}ms`,async()=>{
 const out=output(`clock-${advance}`),realNow=Date.now;let clock=realNow(),calls=0;Date.now=()=>clock;let result:any;
 try{result=await runStudy(plan(),out,options(async(_url,init)=>{calls++;clock+=advance;return response(init);}));}finally{Date.now=realNow;}
 assert.equal(calls,1);assert.equal(result.rows.length,0);assert.equal(result.guard.totals.actualStartedJev,1);
 assert.equal(result.completedAllCalls,false);assert.equal(result.guard.totals.usageUnknownCalls,1);
 assert.equal(result.stop,advance===10000?'timeout':'absolute_wall_deadline');assert.equal(result.guard.missingCallIds.length,84);
 console.log(JSON.stringify({scenario:`clock-${advance}`,calls,stop:result.stop,unknown:result.guard.totals.usageUnknownCalls}));
});
