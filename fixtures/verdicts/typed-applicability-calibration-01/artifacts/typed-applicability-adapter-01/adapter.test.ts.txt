async function setup(options: { clip?: boolean; current?: boolean; companions?: number; padding?: number; title?: string; idWidth?: number; reservedJev?: number; responseStatus?: number } = {}) {
 const manifest = await loadInstalledManifest(root,pin), receipt = resolveReceipt(manifest,'RA03').receipt;
 const artifact = await loadSnapshot(manifest,receipt.proofId), text = snapshotState(artifact,manifest).text;
 const unit = receipt.units[0]!, excerpt = text.slice(unit.start,unit.end);
 const exactClaim = options.current ? 'The latest Fiction prize winner was The Night Watchman.' : 'The 2021 Pulitzer Prize in Fiction winner was The Night Watchman by Louise Erdrich.';
 const source: Evidence = { id:'target',url:receipt.canonicalURL,title:options.title ?? 'Award record',publisher:null,publishedAt:null,
  retrievedAt:'2026-11-14T10:00:00Z',contentBasis:'page_text',excerpt:options.clip ? excerpt.slice(0,80) : excerpt,
  sourceGroup:'pulitzer.org',echoGroup:null,sourceRole:'unknown',directlyAddressesClaim:false,matchesTimePlaceAndPopulation:false,stance:'unrelated',stanceProbability:null };
 const records = [source,...Array.from({length:options.companions ?? 0},(_,i)=>({...source,id:`control-${i}`+"i".repeat(options.idWidth ?? 0),url:`https://control-${i}.example.org/record`,sourceGroup:'example.org',
  excerpt: ('Unrelated control record. '+ 'x'.repeat(options.padding ?? 0)).slice(0,2000) }))];
 const material = captureSelectionMaterial({existing:records,incoming:[],checkedText:exactClaim});
 let clock = Date.parse('2026-11-14T10:00:00Z'), generation = inspectMaterial(material).materialGeneration, install = pin;
 const session = new Session(()=>clock);
 const record: RequestRecord = {input:{schemaVersion:1,sessionId:session.id,sessionEpoch:session.epoch,requestId:'adapter-request',revision:1,
  text:exactClaim,context:[],mode:'typed',language:'en',capturedAt:new Date(clock).toISOString(),capture:{source:'typed_fixture',segmentId:'adapter-fixture',confidence:1},userInitiated:true},
  budget:new AttemptBudget(),controller:new AbortController(),acceptedAt:clock,acceptedMonotonic:0,state:'running',continuationCount:0,results:new Map(),fingerprint:'synthetic-test'};
 session.requests.set(record.input.requestId,record);
 const claim: AcceptedClaim = {id:'adapter-request-claim-0',originalText:exactClaim,checkedText:exactClaim,sourceSpan:{start:0,end:Array.from(exactClaim).length},
  admission:{assertionStatus:'asserted',hasPracticalConsequence:null,matchedSelectedTopic:null,admissionAction:'check',reasonCode:'synthetic-public-test'}};
 const dispatchOwner = captureDispatchOwner(material,{reservedJev:options.reservedJev ?? 0,session,record,claimSlot:0,admittedClaim:claim,readMaterialGeneration:()=>generation,readInstallGeneration:()=>install});
 const owner=inspectDispatchOwner(dispatchOwner).owner;
 let calls=0, lastBody=''; let choice='covered', score=.99, risk='ordinary', intervention=()=>{};
 const fetcher: typeof fetch = async (_url,init) => {
  calls++; if(options.responseStatus) return new Response("transient",{status:options.responseStatus}); lastBody=String(init?.body); const request=JSON.parse(lastBody);
  const answers=Object.fromEntries(Object.entries(request.questions as Record<string,JevQuestion>).map(([key,q])=>{
   if(q.type==='noul') return [key,{noul:.99}];
   const keys=Object.keys(q.criteria);
   const selected = key.endsWith('_applicability') ? choice : key==='claim_risk' ? risk : key==='verdict' ? 'insufficient_evidence'
    :key==='selected_evidence' ? 'none' : key.endsWith('_content_basis') ? 'original_record'
    :key.endsWith('_stance') ? 'supports' : keys.includes('none') ? 'none' : keys[keys.length-1]!;
   const p=key.endsWith('_applicability') ? score:.99;
   return [key,{choice:selected,probabilities:Object.fromEntries(keys.map(k=>[k,k===selected?p:(1-p)/(keys.length-1)]))}];
  }));
  intervention(); return new Response(JSON.stringify({answers,usage:{input_tokens:123}}),{status:200});
 };

 const execute = () => createApplicabilityAdapter({manifest,dispatchOwner,bindings:[{receiptAlias:'RA03',artifact,evidenceId:'target'}],apiKey:'offline-test-key',fetch:fetcher})(claim);
 return {manifest,artifact,material,owner,claim,record,dispatchOwner,execute,stats:()=>({calls,consumed:record.budget.jev,lastBody}),
  intervene:(fn:()=>void)=>{intervention=fn;},advance:()=>{clock+=10001;},cancel:()=>record.controller.abort(),
  replaceMaterial:()=>{generation='new';},replaceInstall:()=>{install='new';},setAnswer:(c:string,p=.99)=>{choice=c;score=p;},setRisk:(v:string)=>{risk=v;}};
}
test('same full judge call binds actual body, validated answers, complete graph and ordinary risk; no core capability',async()=>{
 const s=await setup({companions:5}); const result=await s.execute();
 assert.equal(s.stats().calls,1);assert.equal(s.stats().consumed,1);assert.equal(result.dispatch.questionMap.length,1);
 assert.equal(JSON.parse(s.stats().lastBody).state.evidence.length,6);assert.equal(result.judgment.evidence.length,6);
 assert.equal(result.assessmentContext!.originNodes.length,6);assert.equal(result.assessmentContext!.originLinks.length,15);
 assert.equal(result.eligibility.length,1);const e=result.eligibility[0]!;
 assert.equal(verifySemanticEligibility(e.token,{manifest:s.manifest,material:s.material,owner:s.owner,context:result.assessmentContext!}),true);
 assert.equal(verifySemanticEligibility(structuredClone(e.token),{manifest:s.manifest,material:s.material,owner:s.owner,context:result.assessmentContext!}),false);
 assert.equal(verifyCapability(e.token,{} as never),false);assert.equal(e.audit.recordCapabilityIssued,false);assert.equal(e.audit.temporalAuthorization,false);
 const wire=JSON.parse(s.stats().lastBody);const descriptor=wire.state.record_applicability[0];
 assert.deepEqual(Object.keys(descriptor).sort(),['question','relation','scopeDimensions','snapshotOnly','source'].sort());
 assert.equal('originNodes' in wire.state,false);assert.equal('assessment' in wire.state,false);assert.equal(JSON.stringify(descriptor).includes('Louise'),false);
});
