export interface Plan {

export function validatePlan(plan:Plan){
 check(plan.schemaVersion===1&&plan.kind==='prospective_open_applicability42','wrong_plan');
 check(plan.guard.calls.length===84&&plan.guard.limits.maxCalls===84,'requires_complete84');
 check(plan.guard.blocks.map(b=>b.limits.maxCalls).join(',')==='22,22,20,20','requires_four_blocks');
 check(plan.semanticIds.length===42&&new Set(plan.semanticIds).size===42&&plan.bindingIds.slice().sort().join(',')==='AC-B01,AC-B02','wrong_denominators');
 check(plan.guard.limits.maxInputTokens===1280000&&plan.guard.limits.pendingReserve===24000&&plan.guard.limits.maxRequestBytes===65536&&plan.guard.limits.maxTotalRequestBytes===6291456&&plan.guard.limits.wallMs===120000,'campaign_cap_changed');
 for(const b of plan.guard.blocks)check(b.limits.maxInputTokens===320000&&b.limits.pendingReserve===24000,'block_token_cap_changed');
 check(plan.thresholds.applicability===.85&&plan.thresholds.allProductionGates==='unchanged','threshold_changed');
 for(let i=0;i<42;i++){
  const pair=plan.guard.calls.slice(i*2,i*2+2),id=plan.semanticIds[i]!;
  check(pair.length===2&&pair.every(c=>c.callId.startsWith(id+':'))&&new Set(pair.map(c=>c.callId.split(':')[1])).size===2,'broken_adjacent_pair');
 }
}
export function assertSources(plan:Plan){for(const[path,expected]of Object.entries(plan.sourceHashes))check(hash(readFileSync(path))===expected,`frozen_source_changed:${path}`);}

 check(iadj.differences.length===0&&iadj.authorFreezeUnchanged===true,'identity_labels_unresolved');
 for(const[first,second,count]of [[a,b,36],[ia,ib,8]] as const){check(first.labels.length===count&&second.labels.length===count,'label_count');for(const label of first.labels)check(second.labels.find((x:any)=>x.id===label.id)?.expectedApplicability===label.expectedApplicability,'labels_disagree');}
}

const durable=(path:string,value:unknown)=>writeFileSync(path,JSON.stringify(value,null,2)+'\n',{flush:true});
export async function runStudy(plan:Plan,output:string,options:{apiKey:string;fetch:typeof fetch;transportMode:'live'|'offline-mock'}){
 validatePlan(plan);assertSources(plan);check(!existsSync(output),'fresh_output_required');mkdirSync(output,{recursive:false});
 durable(`${output}/run-start.json`,{planSHA256:hash(JSON.stringify(plan)),transportMode:options.transportMode,actualExaCalls:0,startedAt:new Date().toISOString(),noRetries:true});
 const guard=createStudyGuard(plan.guard,{fetch:options.fetch,onCheckpoint:event=>appendFileSync(`${output}/checkpoints.jsonl`,JSON.stringify(event)+'\n',{flush:true})});
 const contexts=json<{contexts:StudyContext[];bindingControls:any[]}>(plan.contextsPath);
 const manifest=await loadInstalledManifest('/workspace/task-evidence/record-capability-prototype-02/installed','7ba7660e882afd1d5cea4ba37784567cde6ba53dc7633eaa065632f79434fcd4');
 const bindings=[];
 for(const unit of contexts.bindingControls){const receipt=resolveReceipt(manifest,unit.receiptAlias).receipt,artifact=await loadSnapshot(manifest,receipt.proofId);let reason='';try{bindExcerpt(receipt,snapshotState(artifact,manifest).text,unit.evidence.excerpt);}catch(e){reason=e instanceof Error?e.message:'unavailable';}check(reason,'binding_control_unexpectedly_available');bindings.push({id:unit.id,available:false,reason,actualJevAttempts:0,actualExaAttempts:0});}
 durable(`${output}/offline-binding-controls.json`,bindings);
 const rows:any[]=[];let localStop:string|null=null;
 outer:for(const context of contexts.contexts){
  const unit=context.unit,session=new Session(Date.now),at=session.now();
  const claim:AcceptedClaim={id:`${unit.id}-claim-0`,originalText:unit.exactClaimText,checkedText:unit.exactClaimText,sourceSpan:{start:0,end:Array.from(unit.exactClaimText).length},admission:{assertionStatus:'asserted',hasPracticalConsequence:null,matchedSelectedTopic:null,admissionAction:'check',reasonCode:'public-open-calibration'}};
  const record:RequestRecord={input:{schemaVersion:1,sessionId:session.id,sessionEpoch:session.epoch,requestId:unit.id,revision:1,text:unit.exactClaimText,context:[],mode:'typed',language:'en',capturedAt:new Date(at).toISOString(),capture:{source:'typed_fixture',segmentId:'open-calibration',confidence:1},userInitiated:true},budget:new AttemptBudget(),controller:new AbortController(),acceptedAt:at,acceptedMonotonic:0,state:'running',continuationCount:0,results:new Map(),fingerprint:'open-calibration'};session.requests.set(unit.id,record);
  const material=captureSelectionMaterial({existing:context.records,incoming:[],checkedText:unit.exactClaimText}),generation=inspectMaterial(material).materialGeneration;
  const artifact=await loadSnapshot(manifest,resolveReceipt(manifest,unit.receiptAlias).receipt.proofId);
  for(const arm of context.arms){
   const callId=`${unit.id}:${arm}`;
   try{
    assertSources(plan);
    const beforeTokens=record.budget.inputTokens,beforeAttempts=record.budget.jev,beforeObservedTokens=guard.snapshot().totals.measuredInputTokens;
    const result=await guard.run(callId,async lease=>{
     lease.attachAbortController(record.controller);
     const owner=captureDispatchOwner(material,{session,record,claimSlot:0,admittedClaim:claim,readMaterialGeneration:()=>generation,readInstallGeneration:()=> 'study-installed-v1',reservedJev:0});
     const result=await createApplicabilityAdapter({manifest,dispatchOwner:owner,semanticAsOf:unit.asOf,bindings:arm==='applicability'?[{receiptAlias:unit.receiptAlias,artifact,evidenceId:unit.evidence.id}]:[],apiKey:options.apiKey,fetch:lease.fetch})(claim);
     lease.assertOpen();assertSources(plan);
     check(record.budget.jev-beforeAttempts===1,'runtime_attempt_mismatch');
     check(record.budget.inputTokens-beforeTokens===guard.snapshot().totals.measuredInputTokens-beforeObservedTokens,'runtime_usage_mismatch');
     check(result.completedJudgment&&result.assessmentContext&&verifyCompletedJudgment(result.completedJudgment.token,{material,dispatchOwner:owner,context:result.assessmentContext}),'completed_actual_ask_proof_missing');
     for(const e of result.eligibility)check(verifySemanticEligibility(e.token,{manifest,material,owner:(await import('../typed-applicability-adapter-06/runtime-owner.js')).inspectDispatchOwner(owner).owner,context:result.assessmentContext}),'semantic_token_binding_failed');
     const policy=decideVerdict(result.judgment),answers=result.assessmentContext.assessment.state.answers;
     return {callId,unitId:unit.id,arm,cohort:context.cohort,category:unit.category,sourceFamily:unit.sourceFamily,targetId:unit.evidence.id,targetPosition:context.targetPosition,
      judgment:result.judgment,policy,answers,unavailable:result.unavailable,dispatch:result.dispatch,
      completedActualAsk:result.completedJudgment.audit,eligibility:result.eligibility.map(e=>e.audit),
      contextSHA256:hash(JSON.stringify(result.assessmentContext)),nativeGraph:result.assessmentContext.originGraphState,
      runtimeChargedAttempts:record.budget.jev-beforeAttempts,runtimeMeasuredInputTokens:record.budget.inputTokens-beforeTokens};
    });
    if(result.status==='completed'){rows.push(result.value);durable(`${output}/${String(rows.length).padStart(2,'0')}-${unit.id}-${arm}.json`,result.value);}
    else{localStop=result.terminal.failure?.code??'guard_failed';durable(`${output}/failed-terminal.json`,result.terminal);break outer;}
    durable(`${output}/partial-report.json`,{transportMode:options.transportMode,rows,guard:guard.snapshot(),offlineBindingControls:bindings});
   }catch(error){localStop=error instanceof Error?error.message:'local_stop';durable(`${output}/local-stop.json`,{code:localStop});break outer;}
  }

  planSHA256:hash(JSON.stringify(plan)),completedAllCalls:snapshot.completedCallIds.length===84,stop:localStop??snapshot.primaryFailure?.code??null,guard:snapshot,rows,offlineBindingControls:bindings};

}
