const output=()=>join(mkdtempSync(join(tmpdir(),'applicability84-offline-')),'run');
const mock=(usage:boolean=true,status=200):typeof fetch=>async(_url,init)=>{
 if(status!==200)return new Response('terminal-provider-error',{status});
 const req=JSON.parse(String(init?.body));const answers=Object.fromEntries(Object.entries(req.questions).map(([key,value])=>{const q=value as any;if(q.type==='noul')return[key,{noul:.99}];const keys=Object.keys(q.criteria);const choice=key==='claim_risk'?'ordinary':key==='selected_evidence'?'none':key==='verdict'?'insufficient_evidence':key.endsWith('_applicability')?'uncertain':key.endsWith('_content_basis')?'original_record':key.endsWith('_stance')?'unrelated':keys[keys.length-1]!;return[key,{choice,probabilities:Object.fromEntries(keys.map(k=>[k,k===choice?.99:.01/(keys.length-1)]))}];}));
 return new Response(JSON.stringify({answers,...(usage?{usage:{input_tokens:1000}}:{})}),{status:200});
};
test('complete84 real-adapter offline pipeline preserves42labels/2offlinecontrols,84actualmockattempts and all gates',async()=>{
 const plan=preparePlan(),dir=output();let calls=0;const fake=mock();
 const report=await runStudy(plan,dir,{apiKey:'OFFLINE_MOCK_ONLY',transportMode:'offline-mock',fetch:async(...args)=>{calls++;assert.ok(readFileSync(`${dir}/checkpoints.jsonl`,'utf8').includes('request_prepared'));return fake(...args);}});
 assert.equal(calls,84);assert.equal(report.completedAllCalls,true);assert.equal(report.projectProviderCalls,0);assert.equal(report.guard.totals.actualStartedJev,84);assert.equal(report.runtimeLedger.consumedJevAttempts,84);assert.equal(report.guard.totals.measuredInputTokens,84000);assert.equal(report.runtimeLedger.measuredInputTokens,84000);assert.equal(report.guard.totals.usageUnknownCalls,0);
 assert.equal(report.offlineBindingControls.length,2);assert.ok(report.offlineBindingControls.every(c=>c.actualJevAttempts===0));
 assert.equal(report.scoring.byArm.applicability.declared,42);assert.equal(report.scoring.byArm.applicability.applicabilityQuestionAsked,37);assert.equal(report.scoring.byArm.baseline_anchor.rawTernaryExact,null);assert.equal(report.scoring.primaryDecidedCandidate.declared,36);assert.equal(report.scoring.uncertainSafetySentinels.declared,6);assert.equal(report.scoring.selectedCitationAuditInputs.length,0);
 assert.ok(report.rows.every(r=>r.runtimeChargedAttempts===1&&r.completedActualAsk.authorityGranted===false));
 assert.ok(existsSync(`${dir}/report.json`));const checkpoints=readFileSync(`${dir}/checkpoints.jsonl`,'utf8').trim().split('\n').map(l=>JSON.parse(l));assert.equal(checkpoints.filter(c=>c.type==='request_prepared').length,84);assert.equal(checkpoints.find(c=>c.type==='request_prepared').snapshot.totals.actualStartedJev,0);
});
