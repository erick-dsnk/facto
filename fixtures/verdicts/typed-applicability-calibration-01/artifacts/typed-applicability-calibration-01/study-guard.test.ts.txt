import test from 'node:test';
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { createStudyGuard as createGuard, StudyGuardError, type StudyGuardPlan, type StudyLease, type GuardCheckpoint } from './study-guard.ts';

// Tests always inject local transports and a synchronous observation sink.
function createStudyGuard(p: StudyGuardPlan, options: Omit<Parameters<typeof createGuard>[1], 'onCheckpoint'> & {
  onCheckpoint?: (event: GuardCheckpoint) => void;
}) { return createGuard(p, { onCheckpoint: () => undefined, ...options }); }

const url = 'https://api.typesafe.ai/v1/systemone';
const body = JSON.stringify({ model: 'offline', state: { value: 'bounded' }, questions: {} });
const bodyHash = createHash('sha256').update(body).digest('hex');
const bytes = Buffer.byteLength(body);
function plan(counts = [2, 2]): StudyGuardPlan {
  return {
    calls: counts.flatMap((count, b) => Array.from({ length: count }, (_, n) => ({
      callId: `b${b}-c${n}`, blockId: `b${b}`, requestSHA256: bodyHash, requestBytes: bytes,
      denominatorClass: ['positive', 'negative', 'uncertain'][n % 3],
    }))),
    blocks: counts.map((count, n) => ({ blockId: `b${n}`, limits: {
      maxCalls: count, maxInputTokens: 100000, pendingReserve: 24000,
      maxTotalRequestBytes: count * bytes, maxTotalResponseBytes: 128000 * count,
    } })),
    limits: { maxCalls: counts.reduce((a, b) => a + b, 0), maxInputTokens: 1000000,
      pendingReserve: 24000, maxRequestBytes: 65536, maxTotalRequestBytes: counts.reduce((a, b) => a + b, 0) * bytes,
      maxResponseBytesPerCall: 128000, maxTotalResponseBytes: 128000 * counts.reduce((a, b) => a + b, 0), wallMs: 120000 },
  };
}
function response(tokens: unknown = 10): Response { return Response.json({ usage: { input_tokens: tokens }, answers: {} }); }
const fetcher = (fn: (...args: Parameters<typeof fetch>) => Promise<Response> | Response): typeof fetch => async (...args) => fn(...args);
const runWork = async (lease: StudyLease) => (await lease.fetch(url, { method: 'POST', body })).json();
const tick = () => new Promise<void>(resolve => setImmediate(resolve));
async function settles<T>(promise: Promise<T>): Promise<T> {
  let timer: ReturnType<typeof setTimeout>;
  const fail = new Promise<never>((_resolve, reject) => { timer = setTimeout(() => reject(new Error('guard did not settle')), 150); });
  try { return await Promise.race([promise, fail]); } finally { clearTimeout(timer!); }
}
function manualClock() {
  let time = 1000, fire: (() => void) | undefined;
  return { now: () => time, setTimer: (f: () => void) => { fire = f; return 1; }, clearTimer: () => undefined,
    deadline: () => { time += 120000; fire!(); }, advance: (ms: number) => { time += ms; } };
}

test('complete84 inventory executes22/22/20/20 once, counts actual bodies and declared classes', async () => {
  const p = plan([22, 22, 20, 20]); let actual = 0; const events: GuardCheckpoint[] = [];
  const guard = createStudyGuard(p, { fetch: fetcher((_url, init) => { actual++; assert.equal(init?.redirect, 'error'); return response(1); }), onCheckpoint: e => events.push(e) });
  for (const c of p.calls) assert.equal((await guard.run(c.callId, runWork)).status, 'completed');
  const s = guard.snapshot();
  assert.equal(actual, 84); assert.equal(s.status, 'completed'); assert.equal(s.totals.actualStartedJev, 84);
  assert.equal(s.totals.actualStartedExa, 0); assert.equal(s.totals.measuredInputTokens, 84);
  assert.equal(s.totals.requestBytes, 84 * bytes); assert.equal(s.missingCallIds.length, 0);
  assert.equal(s.completeAccuracyClaimPermitted, true);
  assert.deepEqual(Object.values(s.perBlock).map(b => b.actualStartedJev), [22, 22, 20, 20]);
  assert.equal(events.filter(e => e.type === 'request_started').length, 84);
  assert.equal(events.filter(e => e.type === 'request_prepared').length, 84);
  assert.ok(events.filter(e => e.type === 'request_started').every(e => e.requestSHA256 === bodyHash && e.requestBytes === bytes));
  assert.equal(events.filter(e => e.type === 'call_terminal').length, 84);
  await assert.rejects(guard.run(p.calls[0]!.callId, runWork), StudyGuardError);
});

test('response overrun counts the received chunk, aborts actual signal and detaches cancellation', async () => {
  const p = plan(); p.limits.maxResponseBytesPerCall = 10;
  let cancelled = 0, signal: AbortSignal | null | undefined;
  const guard = createStudyGuard(p, { fetch: fetcher((_url, init) => {
    signal = init?.signal;
    return new Response(new ReadableStream({ start(c) { c.enqueue(new Uint8Array(11)); }, cancel() { cancelled++; return new Promise(() => undefined); } }));
  }) });
  const r = await settles(guard.run('b0-c0', runWork));
  assert.equal(r.status, 'failed'); assert.equal(guard.snapshot().totals.responseBytes, 11);
  assert.equal(guard.snapshot().primaryFailure?.code, 'response_byte_cap');
  assert.equal(signal?.aborted, true); assert.equal(cancelled, 1);
});

test('block aggregate response bytes cannot borrow another block response budget', async () => {
  const p = plan(); const wireBytes = Buffer.byteLength(JSON.stringify({ usage: { input_tokens: 1 }, answers: {} }));
  p.blocks[0]!.limits.maxTotalResponseBytes = wireBytes + 1;
  const guard = createStudyGuard(p, { fetch: fetcher(() => response(1)) });

  assert.equal(guard.snapshot().primaryFailure?.code, 'response_byte_cap');

});

test('hung fetch ignoring abort settles on campaign timer and aborts runtime owner', async () => {
  const clock = manualClock(); let signal: AbortSignal | null | undefined;
  const owner = new AbortController();
  const guard = createStudyGuard(plan(), { ...clock, fetch: fetcher((_url, init) => { signal = init?.signal; return new Promise(() => undefined); }) });
  const pending = guard.run('b0-c0', async l => { l.attachAbortController(owner); return runWork(l); });
  await tick(); clock.deadline();
  assert.equal((await settles(pending)).status, 'failed'); assert.equal(signal?.aborted, true);
  assert.equal(owner.signal.aborted, true); assert.equal(guard.snapshot().primaryFailure?.kind, 'deadline');
  assert.equal(guard.snapshot().unknownUsage, true);
});

test('hung reader and hanging cancel cannot hold campaign finalization open', async () => {
  const clock = manualClock(); let cancelled = 0;
  const guard = createStudyGuard(plan(), { ...clock, fetch: fetcher(() => ({ status: 200, statusText: 'OK', headers: new Headers(), body: { getReader: () => ({ read: () => new Promise(() => undefined), cancel: () => { cancelled++; return new Promise(() => undefined); } }) } }) as Response) });
  const pending = guard.run('b0-c0', runWork); await tick(); clock.deadline();
  assert.equal((await settles(pending)).status, 'failed'); assert.ok(cancelled >= 1);
  assert.equal(guard.snapshot().primaryFailure?.code, 'absolute_wall_deadline');
});

test('late headers after deadline are cancelled without mutating finalized accounting', async () => {
  const clock = manualClock(); let resolveResponse!: (r: Response) => void, cancelled = 0;
  const guard = createStudyGuard(plan(), { ...clock, fetch: fetcher(() => new Promise<Response>(resolve => { resolveResponse = resolve; })) });
  const pending = guard.run('b0-c0', runWork); await tick(); clock.deadline(); await settles(pending);
  const before = guard.snapshot();
  resolveResponse(new Response(new ReadableStream({ cancel() { cancelled++; return new Promise(() => undefined); } })));
  await tick(); assert.equal(cancelled, 1); assert.deepEqual(guard.snapshot(), before);
});

test('late EOF cannot introduce usage or a completed answer after timeout', async () => {
  const clock = manualClock(); let end!: (v: ReadableStreamReadResult<Uint8Array>) => void, reads = 0;
  const chunk = new TextEncoder().encode(JSON.stringify({ usage: { input_tokens: 99 }, answers: {} }));
  const reader = { read: () => ++reads === 1 ? Promise.resolve({ done: false, value: chunk }) : new Promise<ReadableStreamReadResult<Uint8Array>>(resolve => { end = resolve; }), cancel: () => new Promise(() => undefined) };
  const guard = createStudyGuard(plan(), { ...clock, fetch: fetcher(() => ({ status: 200, statusText: 'OK', headers: new Headers(), body: { getReader: () => reader } }) as Response) });
  const pending = guard.run('b0-c0', runWork); await tick(); clock.deadline(); await settles(pending);
  const before = guard.snapshot(); end({ done: true, value: undefined }); await tick();
  assert.equal(before.totals.measuredInputTokens, 0); assert.deepEqual(guard.snapshot(), before);
});

test('chunk received after logical deadline but before timer firing is counted before rejection', async () => {
  const clock = manualClock(); const chunk = new Uint8Array(12);
  const guard = createStudyGuard(plan(), { ...clock, fetch: fetcher(() => ({ status: 200, statusText: 'OK', headers: new Headers(), body: { getReader: () => ({ read: async () => { clock.advance(120000); return { done: false, value: chunk }; }, cancel: async () => undefined }) } }) as Response) });
  await guard.run('b0-c0', runWork);
  assert.equal(guard.snapshot().primaryFailure?.kind, 'deadline');
  assert.equal(guard.snapshot().totals.responseBytes, 12);
});

test('validated pre-dispatch checkpoint completes before HTTP and starts no call if storage fails', async () => {
  let actual = 0, prepared = false;
  const guard = createStudyGuard(plan(), { fetch: fetcher(() => { actual++; return response(); }), onCheckpoint(e) {

    assert.equal(e.snapshot.totals.actualStartedJev, 0);

  } });
  assert.equal((await guard.run('b0-c0', runWork)).status, 'failed');
  assert.equal(prepared, true); assert.equal(actual, 0);
  assert.equal(guard.snapshot().totals.actualStartedJev, 0);
  assert.equal(guard.snapshot().unknownUsage, false);
  assert.equal(guard.snapshot().primaryFailure?.code, 'checkpoint_failed');
});
