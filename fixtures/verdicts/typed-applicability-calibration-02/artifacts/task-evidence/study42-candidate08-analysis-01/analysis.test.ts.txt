import test from 'node:test';
import assert from 'node:assert/strict';
import {analyze,hash,type Plan,type Report,type Label,type Capture,type Context}from'./analysis.js';

function fixture(){
 const labels:Label[]=[];
 for(const [cohort,counts]of [['original36',[15,13,6]],['identity8',[3,5,0]]]as const){let index=0;for(const [k,expected]of(['covered','not_covered','uncertain']as const).entries())for(let n=0;n<counts[k]!;n++)labels.push({id:`${cohort}-${index++}`,cohort,category:'synthetic_control',sourceFamily:`${cohort}-family-${n%3}`,expected,excludeFromPrimary:expected==='uncertain'});}
 const unavailable=new Set([...labels.filter(l=>l.cohort==='original36'&&l.expected==='not_covered').slice(0,3),...labels.filter(l=>l.cohort==='original36'&&l.expected==='uncertain').slice(0,1),...labels.filter(l=>l.cohort==='identity8'&&l.expected==='not_covered').slice(0,1)].map(l=>l.id));
 const contexts:Context[]=labels.map(l=>({unit:{id:l.id,exactClaimText:'Synthetic declared proposition '+l.id,evidence:{id:l.id+'-evidence',excerpt:'Synthetic complete bounded record '+l.id}}}));
 const captures:Capture[]=contexts.map(c=>{
  const id=c.unit.id,question=unavailable.has(id)?[]:['a0_applicability'];
  const body=JSON.stringify({state:{checked_proposition:c.unit.exactClaimText,evidence:[{id:c.unit.evidence.id,url:'https://example.org/'+id,excerpt:c.unit.evidence.excerpt}]},questions:{}});
  return{callId:id+':candidate08',unitId:id,arm:'candidate08',addedQuestions:question,preflightUnavailable:unavailable.has(id)?[{evidenceId:c.unit.evidence.id,reason:'current_version_proof_unavailable'}]:[],body,requestSHA256:hash(body),requestBytes:Buffer.byteLength(body)};
 });
 const plan:Plan={denominatorLabels:labels,semanticIds:labels.map(l=>l.id),bindingIds:['AC-B01','AC-B02'],thresholds:{applicability:.85,allProductionGates:'unchanged'},guard:{calls:captures.map(c=>({callId:c.callId,blockId:'block1',requestSHA256:c.requestSHA256,requestBytes:c.requestBytes}))}};
 const rows=labels.map((l,i)=>{const c=contexts[i]!,cap=captures[i]!,id=c.unit.evidence.id;const source={id,url:'https://example.org/'+l.id,title:'Synthetic',excerpt:c.unit.evidence.excerpt,sourceGroup:'example.org',echoGroup:null,directlyAddressesClaim:true,matchesTimePlaceAndPopulation:true,stance:'supports',stanceProbability:.99};
  return{callId:l.id+':candidate08',unitId:l.id,arm:'candidate08',cohort:l.cohort,category:l.category,sourceFamily:l.sourceFamily,targetId:id,
   dispatch:{requestSHA256:cap.requestSHA256,questionMap:cap.addedQuestions.map(q=>({evidenceId:id,questionId:q}))},eligibility:[],unavailable:cap.preflightUnavailable,
   answers:unavailable.has(l.id)?{}:{a0_applicability:{choice:l.expected,probabilities:{covered:l.expected==='covered'?.9:.05,not_covered:l.expected==='not_covered'?.9:.05,uncertain:l.expected==='uncertain'?.9:.05}}},
   judgment:{checkedText:c.unit.exactClaimText,label:'insufficient_evidence',probabilities:{insufficient_evidence:1},selectedEvidenceIds:[],evidence:[source],sourceAssessments:{[id]:{contentBasis:'original_record',confidence:.99}},claimRisk:{level:'ordinary',confidence:.99},originAssessments:[]},policy:{verdict:'insufficient_evidence',selectedEvidenceIds:[]}};
 });
 const report:Report={planSHA256:hash(JSON.stringify(plan)),rows,guard:{completedCallIds:plan.guard.calls.map(c=>c.callId)},offlineBindingControls:[{id:'AC-B01',passed:true},{id:'AC-B02',passed:true}]};
 return{plan,report,captures,contexts,labels,unavailable};
}
const run=(f:ReturnType<typeof fixture>)=>analyze(f.plan,f.report,f.captures,f.contexts);
test('complete candidate-only42 has baseline N/A, never42phantom missing and neverpaired',()=>{const r=run(fixture());assert.deepEqual(r.baseline,{status:'not_run',declaredCalls:0,missingCalls:0,metrics:null});assert.equal(r.pairedComparisons,null);assert.equal(r.previous07Comparison,null);assert.equal(r.candidateInventory.declared,42);assert.equal(r.candidateInventory.completedResults,42);assert.equal(r.byCohort.original36!.declared,34);assert.equal(r.byCohort.identity8!.declared,8);assert.equal(r.uncertainSafetySentinels.declared,6);assert.equal(r.candidateInventory.plannedNoItemIds.length,5);assert.equal(r.candidateInventory.observedNoItemIds.length,5);assert.equal(r.candidateInventory.rawTernaryObserved,37);assert.equal(r.candidateInventory.allDeclaredRawAnswersAvailable,false);});
test('missing positive, negative and uncertain preserve42class denominators including fiveplanned noitems',()=>{const f=fixture();const removed=['covered','not_covered','uncertain'].map(e=>f.labels.find(l=>l.expected===e&&!f.unavailable.has(l.id))!.id);f.report.rows=f.report.rows.filter(r=>!removed.includes(r.unitId));const r=run(f);assert.equal(r.candidateInventory.declared,42);assert.deepEqual(r.candidateInventory.declaredByExpected,{covered:18,not_covered:18,uncertain:6});assert.deepEqual(r.candidateInventory.missingByExpected,{covered:1,not_covered:1,uncertain:1});assert.equal(r.completedGuardCallsWithoutDurableResult.length,3);assert.equal(r.candidateInventory.allDeclaredResultsPresent,false);assert.equal(r.candidateInventory.plannedNoItemIds.length,5);});
test('noitem, itemmissinganswer, invalidanswer and rawuncertain remain different states',()=>{const f=fixture();const available=f.report.rows.filter(r=>r.dispatch.questionMap.length);delete available[0]!.answers.a0_applicability;available[1]!.answers.a0_applicability={choice:'covered',probabilities:{covered:NaN}};available[2]!.answers.a0_applicability={choice:'uncertain',probabilities:{covered:.05,not_covered:.05,uncertain:.9}};const r=run(f);assert.equal(r.candidateInventory.states.no_item,5);assert.equal(r.candidateInventory.states.item_missing_answer,1);assert.equal(r.candidateInventory.states.invalid_answer,1);assert.equal(r.candidateInventory.allPlannedAnswersValid,false);assert.ok(r.diagnostics.some(x=>x.state==='asked'&&x.rawAnswer.choice==='uncertain'));});
test('non-grant enum disagreements differ from highcovered safety failure and actual typedgrant',()=>{const f=fixture();const negative=f.report.rows.filter(r=>f.labels.find(l=>l.id===r.unitId)!.expected==='not_covered'&&r.dispatch.questionMap.length);negative[0]!.answers.a0_applicability={choice:'uncertain',probabilities:{covered:.02,not_covered:.03,uncertain:.95}};negative[1]!.answers.a0_applicability={choice:'covered',probabilities:{covered:.9,not_covered:.05,uncertain:.05}};negative[1]!.judgment.claimRisk={level:'health_safety',confidence:.99};const r=run(f);assert.deepEqual(r.candidateInventory.nonGrantNotCoveredUncertainDisagreements,[negative[0]!.unitId]);assert.deepEqual(r.candidateInventory.thresholdedCovered.negativeAccepted,[negative[1]!.unitId]);assert.equal(r.candidateInventory.typedEligibility.negativeIssued.length,0);assert.equal(r.factualCoverageCredit,0);});
test('selectedcovered threshold stays exact.85 and rawcovered belowgate stays observed',()=>{const f=fixture();const rows=f.report.rows.filter(r=>r.dispatch.questionMap.length);rows[0]!.answers.a0_applicability={choice:'covered',probabilities:{covered:.8499999,not_covered:.1,uncertain:.0500001}};rows[1]!.answers.a0_applicability={choice:'covered',probabilities:{covered:.85,not_covered:.1,uncertain:.05}};const r=run(f);assert.equal(r.diagnostics.find(d=>d.id===rows[0]!.unitId)!.coveredAt85,false);assert.equal(r.diagnostics.find(d=>d.id===rows[1]!.unitId)!.coveredAt85,true);});
test('undeclared baseline or duplicate report row is rejected',()=>{let f=fixture();f.report.rows.push({...f.report.rows[0],arm:'baseline_anchor',callId:f.labels[0]!.id+':baseline_anchor'});assert.throws(()=>run(f),/undeclared/);f=fixture();f.report.rows.push(f.report.rows[0]);assert.throws(()=>run(f),/duplicate/);});
test('changed request source bytes or targetidentity are rejected',()=>{let f=fixture();f.report.rows[0]!.judgment.evidence[0]!.excerpt='silently changed';assert.throws(()=>run(f),/material_changed/);f=fixture();f.report.rows[0]!.targetId='other';assert.throws(()=>run(f),/target_or_claim_changed/);});
test('no-item target cannot hide a typed grant in totals',()=>{const f=fixture();const row=f.report.rows.find(r=>!r.dispatch.questionMap.length)!;row.eligibility.push({evidenceId:row.targetId}as never);const r=run(f);assert.deepEqual(r.candidateInventory.typedEligibility.inconsistentWithCoveredAnswer,[row.unitId]);});
test('hypothetical definitive output retains exact selectedcitation audit but receives no factualcredit',()=>{const f=fixture();const row=f.report.rows[0]!;row.policy={verdict:'supported',selectedEvidenceIds:[row.targetId]};const r=run(f);assert.equal(r.selectedCitationAuditInputs.length,1);assert.equal(r.selectedCitationAuditInputs[0]!.selectedSources[0]!.excerpt,row.judgment.evidence[0]!.excerpt);assert.equal(r.factualCoverageCredit,0);});
test('observed answer before accountingfailure remains unscored and does not remove missingcase',()=>{const f=fixture(),row=f.report.rows.shift()!;f.report.guard.completedCallIds=f.report.guard.completedCallIds.filter(id=>id!==row.callId);const observed={callId:row.callId,answers:row.answers,dispatch:row.dispatch};const r=analyze(f.plan,f.report,f.captures,f.contexts,[observed]);assert.equal(r.candidateInventory.missingResults,1);assert.deepEqual(r.observedButUnscored,[observed]);assert.equal(r.candidateInventory.rawTernaryObserved,36);});
