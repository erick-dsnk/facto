import { test } from 'node:test';
import assert from 'node:assert/strict';
import fs from 'node:fs';
import { createHash } from 'node:crypto';
import { join } from 'node:path';
import { runStudy, assertSources, validatePlan, type Plan } from '/workspace/task-evidence/typed-applicability-calibration-02/runner.js';

const root = '/workspace/task-evidence/study42-corrected08-runtime-review';
const planPath = '/workspace/task-evidence/typed-applicability-calibration-02/corrected08-42-plan.json';
const hash = (value: string | Uint8Array) => createHash('sha256').update(value).digest('hex');
globalThis.fetch = async () => { throw new Error('offline_global_fetch_forbidden'); };
function plan(): Plan {
  const value = JSON.parse(fs.readFileSync(planPath, 'utf8'));
  assert.equal(hash(JSON.stringify(value)), '699582a0b64569f73b20376521be7aa6bfc4b1439d06ae6182e5942b21ab5de3');
  validatePlan(value); assertSources(value); return value;
}
const options = (fetch: typeof globalThis.fetch) => ({ apiKey: 'OFFLINE_INJECTED', fetch, transportMode: 'offline-mock' as const });
function response(init?: RequestInit) {
  const request = JSON.parse(String(init?.body));
  const answers = Object.fromEntries(Object.entries(request.questions).map(([key, value]) => {
    const q = value as any;
    if (q.type === 'noul') return [key, { noul: .5 }];
    const keys = Object.keys(q.criteria);
    const choice = key === 'claim_risk' ? 'unknown' : key === 'selected_evidence' ? 'none' : key === 'verdict' ? 'insufficient_evidence' : key.endsWith('_applicability') ? 'uncertain' : keys[keys.length - 1]!;
    return [key, { choice, probabilities: Object.fromEntries(keys.map(k => [k, k === choice ? .99 : .01 / (keys.length - 1)])) }];
  }));
  return Response.json({ answers, usage: { input_tokens: 10 } });
}

test('candidate-only runner rejects inventory and source drift before creating live-style output', async () => {
  const changes: [string, (p: Plan) => void][] = [
    ['missing-call', p => { p.guard.calls.pop(); }],
    ['duplicate-call', p => { p.guard.calls[1]!.callId = p.guard.calls[0]!.callId; }],
    ['baseline-arm', p => { p.guard.calls[0]!.callId = p.semanticIds[0] + ':baseline_anchor'; }],
    ['wrong-order', p => { [p.guard.calls[0], p.guard.calls[1]] = [p.guard.calls[1]!, p.guard.calls[0]!]; }],
    ['source-pin', p => { p.sourceHashes[Object.keys(p.sourceHashes)[0]!] = '0'.repeat(64); }],
  ];
  let calls = 0;
  for (const [name, mutate] of changes) {
    const changed = plan(); mutate(changed);
    const output = join(root, `rejected-${name}`); assert.equal(fs.existsSync(output), false);
    await assert.rejects(runStudy(changed, output, options(async () => { calls++; throw new Error('unexpected_mock_dispatch'); })));
    assert.equal(fs.existsSync(output), false);
  }
  assert.equal(calls, 0);
});

test('actual body mismatch stops before HTTP while preserving logical versus actual attempt counts', async () => {
  const changed = plan(); changed.guard.calls[0]!.requestSHA256 = '0'.repeat(64);
  const output = join(root, 'body-mismatch'); assert.equal(fs.existsSync(output), false);
  let calls = 0;
  const result = await runStudy(changed, output, options(async () => { calls++; throw new Error('unexpected_mock_dispatch'); }));
  assert.equal(calls, 0); assert.equal(result.guard.totals.actualStartedJev, 0);
  assert.equal(result.runtimeLedger.consumedJevAttempts, 1);
  assert.equal(result.rows.length, 0); assert.equal(result.guard.missingCallIds.length, 42);
  assert.equal(result.stop, 'request_body_hash_mismatch');
  assert.equal(result.guard.totals.usageUnknownCalls, 0);
});

test('post-response source change retains observed output but accepts no completed row or subsequent call', async () => {
  const changed = plan();
  const watched = join(root, 'reviewer-owned-source-sentinel.txt');
  fs.writeFileSync(watched, 'before', { flag: 'wx' });
  changed.sourceHashes[watched] = hash(fs.readFileSync(watched));
  const output = join(root, 'post-response-source-change'); assert.equal(fs.existsSync(output), false);
  let calls = 0;
  const result = await runStudy(changed, output, options(async (_url, init) => {
    calls++; fs.writeFileSync(watched, 'after'); return response(init);
  }));
  assert.equal(calls, 1); assert.equal(result.guard.totals.actualStartedJev, 1);
  assert.equal(result.runtimeLedger.consumedJevAttempts, 1); assert.equal(result.runtimeLedger.measuredInputTokens, 10);
  assert.equal(result.rows.length, 0); assert.equal(result.guard.missingCallIds.length, 42);
  assert.ok(result.stop); assert.equal(result.guard.status, 'stopped');
  const first = changed.semanticIds[0];
  assert.ok(fs.existsSync(join(output, `observed-${first}-candidate08.json`)));
  assert.ok(fs.existsSync(join(output, 'report.json')));
});
