// Vendor contract: no hidden question ID or any other question is available.
const visible=(request:any,q:any)=>({state:request.state,instructions:q.instructions,criteria:q.criteria});
function resolveVisible(payload:any){
 const source=payload.instructions.match(/use `evidence\[(\d+)\]` \(retained id ("(?:\\.|[^"\\])*")\)/);
 const descriptor=payload.instructions.match(/and `record_applicability\[(\d+)\]` as its bound record descriptor/);
 assert.ok(source&&descriptor,'own instructions must identify both paths without hidden question ID');
 const record=payload.state.evidence[Number(source[1])],entry=payload.state.record_applicability[Number(descriptor[1])];
 assert.ok(record&&entry);assert.equal(record.id,JSON.parse(source[2]));assert.equal(entry.evidenceId,record.id);
 return{record,entry,evidenceIndex:Number(source[1]),descriptorIndex:Number(descriptor[1])};
}
test('07 distinct intended targets produce identical documented per-question payloads;08 makes them distinct',async()=>{
 const old=await capture(7,['RA03','RA01'],['RA01','RA03']),fresh=await capture(8,['RA03','RA01'],['RA01','RA03']);
 const oldItems=Object.values(old.questions).filter((q:any)=>JSON.stringify(q.criteria)===JSON.stringify(APPLICABILITY_CRITERIA));
 const newItems=Object.values(fresh.questions).filter((q:any)=>JSON.stringify(q.criteria)===JSON.stringify(APPLICABILITY_CRITERIA));assert.equal(oldItems.length,2);assert.equal(newItems.length,2);
 assert.deepEqual(visible(old,oldItems[0]),visible(old,oldItems[1]));assert.notDeepEqual(visible(fresh,newItems[0]),visible(fresh,newItems[1]));
 const resolved=newItems.map(q=>resolveVisible(visible(fresh,q)));assert.deepEqual(resolved.map(r=>r.record.id),['source-RA01','source-RA03']);
 assert.deepEqual(resolved.map(r=>[r.evidenceIndex,r.descriptorIndex]),[[1,0],[0,1]]);
 for(const q of newItems as any[]){assert.ok(q.instructions.endsWith(APPLICABILITY_INSTRUCTIONS));assert.deepEqual(q.criteria,APPLICABILITY_CRITERIA);}
 const oldBase=Object.fromEntries(Object.entries(old.questions).filter(([,q]:any)=>JSON.stringify(q.criteria)!==JSON.stringify(APPLICABILITY_CRITERIA)));
 const newBase=Object.fromEntries(Object.entries(fresh.questions).filter(([,q]:any)=>JSON.stringify(q.criteria)!==JSON.stringify(APPLICABILITY_CRITERIA)));
 assert.deepEqual(newBase,oldBase);assert.deepEqual(fresh.state.evidence,old.state.evidence);assert.equal(fresh.state.semantic_as_of,old.state.semantic_as_of);
});
