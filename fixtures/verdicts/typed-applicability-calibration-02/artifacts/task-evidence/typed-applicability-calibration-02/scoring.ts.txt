import {createHash}from'node:crypto';
export interface Denominator{id:string;cohort:string;category:string;sourceFamily:string;expected:'covered'|'not_covered'|'uncertain';excludeFromPrimary:boolean}
const sha=(v:string)=>createHash('sha256').update(v).digest('hex');
const counts=(values:string[])=>Object.fromEntries([...new Set(values)].map(k=>[k,values.filter(v=>v===k).length]));
export function scoreCorrected(labels:Denominator[],rows:any[]){
 const cells=labels.map(label=>{const row=rows.find(r=>r.unitId===label.id),mapping=row?.dispatch.questionMap.find((q:any)=>q.evidenceId===row.targetId),answer=mapping?row.answers?.[mapping.questionId]:undefined;return{...label,row,answer,asked:!!answer,covered:answer?.choice==='covered'&&(answer.probabilities?.covered??0)>=.85,typed:!!row?.eligibility.length};});
 const metric=(subset:typeof cells)=>({declared:subset.length,completed:subset.filter(c=>c.row).length,missing:subset.filter(c=>!c.row).length,declaredByExpected:counts(subset.map(c=>c.expected)),completedByExpected:counts(subset.filter(c=>c.row).map(c=>c.expected)),missingByExpected:counts(subset.filter(c=>!c.row).map(c=>c.expected)),missingIds:subset.filter(c=>!c.row).map(c=>c.id),
  questionAsked:subset.filter(c=>c.asked).length,unaskedOrMissing:subset.filter(c=>!c.asked).length,rawTernaryExact:subset.filter(c=>c.asked&&c.answer.choice===c.expected).length,rawTernaryDeclaredDenominator:subset.length,rawTernaryObservedDenominator:subset.filter(c=>c.asked).length,
  combinedAvailabilityAndCoverageGate:{positiveDeclared:subset.filter(c=>c.expected==='covered').length,positiveAccepted:subset.filter(c=>c.expected==='covered'&&c.covered).length,negativeDeclared:subset.filter(c=>c.expected==='not_covered').length,negativeAccepted:subset.filter(c=>c.expected==='not_covered'&&c.covered).length,uncertainDeclared:subset.filter(c=>c.expected==='uncertain').length,unsafeUncertainGrants:subset.filter(c=>c.expected==='uncertain'&&c.covered).length},
  privateTypedEligibility:{issued:subset.filter(c=>c.typed).length,notCoveredGrants:subset.filter(c=>c.expected==='not_covered'&&c.typed).length,uncertainGrants:subset.filter(c=>c.expected==='uncertain'&&c.typed).length},
  policyDecisions:counts(subset.filter(c=>c.row).map(c=>c.row.policy.verdict)),guardUnavailable:subset.filter(c=>c.row&&!c.asked).map(c=>({id:c.id,reasons:c.row.unavailable}))});
 const by=(key:'cohort'|'category'|'sourceFamily')=>Object.fromEntries([...new Set(labels.map(l=>l[key]))].map(k=>[k,metric(cells.filter(c=>c[key]===k))]));
 return{studyKind:'corrected08_candidate_only42',declaredSemanticUnits:labels.length,declaredCandidateCalls:labels.length,baselineComparison:'not_declared_not_estimated',historical07Pooling:false,
  all:metric(cells),byCohort:by('cohort'),byCategory:by('category'),bySourceFamily:by('sourceFamily'),primaryDecided:metric(cells.filter(c=>!c.excludeFromPrimary&&c.expected!=='uncertain')),uncertainSentinels:metric(cells.filter(c=>c.expected==='uncertain')),
  rows:cells.map(c=>({id:c.id,cohort:c.cohort,expected:c.expected,category:c.category,sourceFamily:c.sourceFamily,completed:!!c.row,questionAsked:c.asked,rawAnswer:c.answer??null,coveredAt85:c.covered,typedEligibility:c.typed,guardUnavailable:c.row?.unavailable??[],targetDiagnostics:c.row?{source:c.row.judgment.evidence.find((e:any)=>e.id===c.row.targetId),basis:c.row.judgment.sourceAssessments[c.row.targetId],risk:c.row.judgment.claimRisk,policy:c.row.policy}:null})),
  selectedCitationAuditInputs:rows.filter(r=>r.policy.verdict!=='insufficient_evidence').map(r=>({unitId:r.unitId,exactClaim:r.judgment.checkedText,claimSHA256:sha(r.judgment.checkedText),policy:r.policy,selectedSources:r.policy.selectedEvidenceIds.map((id:string)=>{const s=r.judgment.evidence.find((e:any)=>e.id===id);return{id,url:s.url,urlSHA256:sha(s.url),excerpt:s.excerpt,excerptSHA256:sha(s.excerpt),sourceGroup:s.sourceGroup,echoGroup:s.echoGroup};}),requiresIndependentWholeQualifierAndLineageAudit:true})),
  factualCoverageCredit:0,productionActivation:false,interpretation:'Distinct candidate-only OPEN development observation; no declared baseline or paired-causal comparison, no pooling with07first84. Raw ternary applicability, deterministic availability, ordinary-risk-gated private eligibility and full unchanged policy outcomes are separate. Missing/guard-excluded/uncertain cases remain declared; labels are not factual verdicts.'};
}
